normalizeChangeSpec[cs_] := 
  Module[{c = cs}, 
   Which[c === None || c === {}, {}, 
    ListQ[c] && Length[c] >= 2 && 
     MemberQ[{"ratio", "absolute"}, 
      c[[2]]] && (StringQ[c[[1]]] || ListQ[c[[1]]]), {c}, True, c]];

processBaseParamsWithNames[paramsSpec_] := 
  Module[{rules = {}, displayNames = {}, symbols = {}}, 
   Which[MatchQ[paramsSpec, {_String, _String, _?NumericQ}], 
    AppendTo[symbols, Symbol[paramsSpec[[1]]]]; 
    AppendTo[displayNames, paramsSpec[[2]]]; 
    rules = {Symbol[paramsSpec[[1]]] -> paramsSpec[[3]]}, 
    MatchQ[paramsSpec, {_String, _?NumericQ}], 
    AppendTo[symbols, Symbol[paramsSpec[[1]]]]; 
    AppendTo[displayNames, paramsSpec[[1]]]; 
    rules = {Symbol[paramsSpec[[1]]] -> paramsSpec[[2]]}, 
    MatchQ[paramsSpec, {{_String, _String, _?NumericQ} ..}], 
    Do[AppendTo[symbols, Symbol[param[[1]]]]; 
     AppendTo[displayNames, param[[2]]]; 
     AppendTo[rules, Symbol[param[[1]]] -> param[[3]]], {param, 
      paramsSpec}], MatchQ[paramsSpec, {{_String, _?NumericQ} ..}], 
    Do[AppendTo[symbols, Symbol[param[[1]]]]; 
     AppendTo[displayNames, param[[1]]]; 
     AppendTo[rules, Symbol[param[[1]]] -> param[[2]]], {param, 
      paramsSpec}], True, 
    Return[{$Failed, $Failed, $Failed}]]; {rules, displayNames, 
    symbols}];

processChangeSpecWithNames[changeSpec_, baseRules_, displayAssoc_] := 
  Module[{results = {}}, 
   Do[Which[
     ListQ[spec] && Length[spec] >= 3 && StringQ[spec[[1]]] && 
      MemberQ[{"ratio", "absolute"}, spec[[2]]] && 
      AllTrue[Drop[spec, 2], NumericQ], 
     Module[{param, method, vals, baseVal}, param = Symbol[spec[[1]]];
       method = spec[[2]]; vals = Drop[spec, 2]; 
      baseVal = param /. baseRules; 
      If[! NumericQ[baseVal], Return[$Failed]]; 
      AppendTo[results, 
       Table[param -> 
         Switch[method, "ratio", baseVal*v, "absolute", v], {v, 
         vals}]];], 
     ListQ[spec] && Length[spec] == 3 && ListQ[spec[[1]]] && 
      MemberQ[{"ratio", "absolute"}, spec[[2]]] && 
      MatrixQ[spec[[3]], NumericQ] && 
      Length[spec[[1]]] == Length[spec[[3, 1]]], 
     Module[{params, method, valueSets}, params = Symbol /@ spec[[1]];
       method = spec[[2]]; valueSets = spec[[3]]; 
      If[! And @@ (NumericQ[# /. baseRules] & /@ params), 
       Return[$Failed]]; 
      AppendTo[results, 
       Table[Table[
         params[[i]] -> 
          Switch[method, 
           "ratio", (params[[i]] /. baseRules)*valueSet[[i]], 
           "absolute", valueSet[[i]]], {i, 
          Length[params]}], {valueSet, valueSets}]];], True, 
     Return[$Failed]], {spec, changeSpec}]; results];

generateParameterCombinationsSimple[baseRules_, changeRules_] := 
  Module[{allCombinations = {}}, 
   If[changeRules === {}, Return[{baseRules}]]; 
   Table[Module[{flatRules = Flatten[combo, 1]}, 
     AppendTo[allCombinations, 
       Join[DeleteCases[baseRules, 
         Rule[p_, _] /; MemberQ[First /@ flatRules, p]], 
        flatRules]];], {combo, Tuples[changeRules]}]; allCombinations];

getMentionedParams[changeSpec_] := 
  Module[{specs, names}, 
   specs = If[ListQ[changeSpec], changeSpec, {changeSpec}]; 
   names = Flatten@
     Map[Function[spec, 
       Which[ListQ[spec] && Length[spec] >= 1 && 
         StringQ[spec[[1]]], {spec[[1]]}, 
        ListQ[spec] && Length[spec] >= 1 && ListQ[spec[[1]]] && 
         AllTrue[spec[[1]], StringQ], spec[[1]], True, {}]], specs]; 
   DeleteDuplicates[Symbol /@ names]];

formatLegendValue[val_] := 
  If[IntegerQ[val] || IntegerPart[val] == val, IntegerPart[val], 
   NumberForm[val, {Infinity, 3}, NumberPadding -> {"", ""}, 
    ExponentFunction -> (Null &)]];

generateLegendsWithNames[expandedRules_, mentionedParams_, 
   displayAssoc_] := 
  Module[{legendForRule}, If[mentionedParams === {}, Return[{}]]; 
   legendForRule[rules_] := 
    Module[{kept, ordered}, 
     kept = Select[rules, MemberQ[mentionedParams, First[#]] &]; 
     ordered = 
      SortBy[kept, (FirstPosition[mentionedParams, 
           First[#], {Infinity}][[1]] &)]; 
     Row[Table[
       With[{sym = First[r], val = Last[r]}, 
        Row[{displayAssoc[sym], " = ", formatLegendValue[val]}]], {r, 
        ordered}], ", "]]; 
   Table[legendForRule[rules], {rules, expandedRules}]];

normalizeExprList[expr_] := If[ListQ[expr], expr, {expr}];

normalizeAxesSpec[axesSpec_, xSym_, exprList_] := 
  Module[{n = Length[exprList]}, 
   Which[axesSpec === Automatic, 
    Table[{ToString[xSym], ToString[HoldForm[exprList[[i]]]]}, {i, 
      n}], axesSpec === None, ConstantArray[None, n], 
    MatchQ[axesSpec, {_, _}] && ! ListQ[First[axesSpec]], 
    ConstantArray[axesSpec, n], 
    ListQ[axesSpec] && Length[axesSpec] == n && 
     AllTrue[axesSpec, MatchQ[#, {_, _}] &], axesSpec, True, 
    Return[$Failed]]];

normalizeLabelSpec[labelSpec_, n_] := 
  Which[labelSpec === None, ConstantArray[None, n], 
   StringQ[labelSpec], ConstantArray[labelSpec, n], 
   ListQ[labelSpec] && Length[labelSpec] == n, labelSpec, 
   True, $Failed];

    processBaseParamsWithNames[paramsSpec];

   displayAssoc = AssociationThread[paramSymbols, displayNames];

     displayAssoc]; If[changeRules === $Failed, Return[$Failed]];

    generateParameterCombinationsSimple[baseRules, changeRules]; 
   If[expandedRules === $Failed, Return[$Failed]]; 
   legends = 
    generateLegendsWithNames[expandedRules, mentionedParams, 
     displayAssoc];

     Which[OptionValue["axes"] === None, ConstantArray[None, n], 
      OptionValue["axes"] === Automatic, ConstantArray[{"", ""}, n], 
      True, normalizeAxesSpec[OptionValue["axes"], x, exprList]], 
     normalizeAxesSpec[OptionValue["axes"], x, exprList]]; 
   If[axesList === $Failed, Return[$Failed]]; 
   labelList = normalizeLabelSpec[OptionValue["label"], n]; 
   If[labelList === $Failed, Return[$Failed]]; 
   dashForExpr[j_Integer] := 
    Which[j == 1, {}, j == 2, Dashed, j == 3, Dotted, j == 4, 
     DotDashed, True, 
     Module[{k = j - 4}, 
      Dashing[{0.018 + 0.006 k, 0.010 + 0.004 k}]]]; 
   colorList = 
    If[OptionValue[PlotStyle] === Automatic, 
     Table[ColorData[97][i], {i, Length[expandedRules]}], 
     OptionValue[PlotStyle]]; 
   If[joinOpt =!= None, 
    Return@Module[{exprCount, joinLabels, axesOne, labelOne, curves, 
       styles, joinLegend, paramLegend, combinedLegend, plotOne}, 
      exprCount = Length[exprList]; 
      joinLabels = 
       Which[StringQ[joinOpt], ConstantArray[joinOpt, exprCount], 
        ListQ[joinOpt] && Length[joinOpt] == exprCount, joinOpt, True,
         Table["f" <> ToString[j], {j, exprCount}]];

       Flatten[Table[(exprList[[j]] /. expandedRules[[k]]), {k,

      styles = 
       Flatten[Table[
         Directive[colorList[[k]], dashForExpr[j]], {k, 
          Length[expandedRules]}, {j, exprCount}], 1]; 
      joinLegend = 
       LineLegend[
        Table[Directive[Black, dashForExpr[j], 
          AbsoluteThickness[1]], {j, exprCount}], joinLabels, 
        LabelStyle -> defaultLegendStyle, LegendLayout -> "Column"]; 
      paramLegend = 
       If[mentionedParams === {} || legends === {} || 
         legends === {{}} || legends === {""}, None, 
        LineLegend[
         Table[Directive[colorList[[k]]], {k, Length[expandedRules]}],
          legends, LabelStyle -> defaultLegendStyle, 
         LegendLayout -> "Column"]]; 
      combinedLegend = 
       Which[paramLegend === None, joinLegend, True, 
        Column[{joinLegend, paramLegend}, Spacings -> 0.6, 
         Alignment -> Left]]; 
      plotOne = 
       If[polarQ, 
        PolarPlot[Evaluate@curves, plotfor, PlotStyle -> styles, 
         PlotRange -> Automatic, BaseStyle -> defaultAxesStyle, 
         AspectRatio -> 0.7, 
         AxesLabel -> 
          If[axesOne === None, None, 
           Map[Style[#, defaultFrameLabelStyle] &, axesOne]], 
         GridLines -> None, 
         PlotLegends -> 
          If[legendPos === None, None, 
           Placed[combinedLegend, legendPos]], 
         ImageSize -> OptionValue[ImageSize], 
         LabelStyle -> OptionValue["LabelStyle"], 
         PlotLabel -> 
          If[labelOne =!= None, 
           Style[labelOne, defaultPlotLabelStyle], None], 
         Background -> White, 
         ImagePadding -> OptionValue[ImagePadding]], 
        Plot[Evaluate@curves, plotfor, PlotStyle -> styles, 
         BaseStyle -> defaultAxesStyle, PlotRange -> Automatic, 
         Frame -> OptionValue[Frame], 
         FrameLabel -> 
          If[OptionValue[Frame] && axesOne =!= None, 
           Map[Style[#, defaultFrameLabelStyle] &, axesOne], None], 
         FrameTicksStyle -> defaultFrameStyle, 
         
         AxesLabel -> 
          If[! OptionValue[Frame] && axesOne =!= None, 
           Map[Style[#, defaultFrameLabelStyle] &, axesOne], None], 
         GridLines -> None, 
         PlotLegends -> 
          If[legendPos === None, None, 
           Placed[combinedLegend, legendPos]], 
         ImageSize -> OptionValue[ImageSize], 
         LabelStyle -> OptionValue["LabelStyle"], 
         PlotLabel -> 
          If[labelOne =!= None, 
           Style[labelOne, defaultPlotLabelStyle], None], 
         Background -> White, FrameStyle -> defaultFrameStyle, 
         ImagePadding -> OptionValue[ImagePadding]]]; 
      If[exportOpt =!= None, 
       Module[{ensureDir, makePath, doExportOne}, 
        ensureDir[p_] := 
         If[! DirectoryQ[p], 
          CreateDirectory[p, CreateIntermediateDirectories -> True]]; 
        makePath[path_] := 
         If[StringTake[path, -1] =!= $PathnameSeparator, 
          path <> $PathnameSeparator, path]; 
        doExportOne[plot_, {path_, name_}] := 
         Module[{p = makePath[path], fp}, ensureDir[p]; 
          fp = p <> name <> ".png"; 
          Export[fp, plot, ImageResolution -> 300, 
           Background -> White]; SystemOpen[fp];]; 
        If[MatchQ[exportOpt, {_String, _String}], 
         doExportOne[plotOne, exportOpt]];]]; plotOne]]; 
   styleList = 
    If[OptionValue[PlotStyle] === Automatic, 
     Table[ColorData[97][i], {i, Length[expandedRules]}], 
     OptionValue[PlotStyle]]; 
   plots = Table[
     allCurvesPerPlot = 
      Table[exprList[[i]] /. rules, {rules, expandedRules}]; 
     If[polarQ, 
      PolarPlot[Evaluate@allCurvesPerPlot, plotfor, 
       PlotStyle -> styleList, PlotRange -> Automatic, 
       BaseStyle -> defaultAxesStyle, AspectRatio -> 0.7,

           LabelStyle -> defaultLegendStyle], legendPos]], 
       ImageSize -> OptionValue[ImageSize], 
       LabelStyle -> OptionValue["LabelStyle"], 
       PlotLabel -> 
        If[labelList[[i]] =!= None, 
         Style[labelList[[i]], defaultPlotLabelStyle], None], 
       Background -> White, 
       ImagePadding -> OptionValue[ImagePadding]], 
      Plot[Evaluate@allCurvesPerPlot, plotfor, PlotStyle -> styleList,
        BaseStyle -> defaultAxesStyle, PlotRange -> Automatic, 
       Frame -> OptionValue[Frame], 
       FrameLabel -> 
        If[OptionValue[Frame] && axesList[[i]] =!= None, 
         Map[Style[#, defaultFrameLabelStyle] &, axesList[[i]]], 
         None], FrameTicksStyle -> defaultFrameStyle, 
       AxesLabel -> 
        If[! OptionValue[Frame] && axesList[[i]] =!= None, 
         Map[Style[#, defaultFrameLabelStyle] &, axesList[[i]]], 
         None], GridLines -> None, 
       PlotLegends -> 
        If[mentionedParams === {} || legendPos === None, None, 
         Placed[LineLegend[styleList, legends, 
           LabelStyle -> defaultLegendStyle], legendPos]], 
       ImageSize -> OptionValue[ImageSize], 
       LabelStyle -> OptionValue["LabelStyle"], 
       PlotLabel -> 
        If[labelList[[i]] =!= None, 
         Style[labelList[[i]], defaultPlotLabelStyle], None], 
       Background -> White, FrameStyle -> defaultFrameStyle, 
       ImagePadding -> OptionValue[ImagePadding]]], {i, n}]; 
   If[exportOpt =!= None, 
    Module[{ensureDir, makePath, doExportOne, combined}, 
     ensureDir[p_] := 
      If[! DirectoryQ[p], 
       CreateDirectory[p, CreateIntermediateDirectories -> True]]; 
     makePath[path_] := 
      If[StringTake[path, -1] =!= $PathnameSeparator, 
       path <> $PathnameSeparator, path]; 
     doExportOne[plot_, {path_, name_}] := 
      Module[{p = makePath[path], fp}, ensureDir[p]; 
       fp = p <> name <> ".png"; 
       Export[fp, plot, ImageResolution -> 300, Background -> White]; 
       SystemOpen[fp];]; 
     Which[MatchQ[exportOpt, {_String, _String}], 
      combined = 
       If[n == 1, First[plots], Column[plots, Spacings -> 1]]; 
      doExportOne[combined, exportOpt], 
      ListQ[exportOpt] && 
       AllTrue[exportOpt, MatchQ[#, {_String, _String}] &] && 
       Length[exportOpt] == n, 
      Do[doExportOne[plots[[i]], exportOpt[[i]]], {i, n}], True, 
      Null];]]; 
   If[n == 1, First[plots], Column[plots, Spacings -> 1]]];

SheetsToPick[filePath_String] := 
  Module[{fullPath, sheets, sheetData, result = {}, colsWithData, 
    allData, rowStart = RowStart}, 
   fullPath = 
    If[StringStartsQ[filePath, "."] || StringStartsQ[filePath, "\\"] ||
       StringStartsQ[filePath, "/"], 
     FileNameJoin[{NotebookDirectory[], filePath}], filePath]; 
   If[! FileExistsQ[fullPath], Return[$Failed]]; 
   sheets = Import[fullPath, "Sheets"]; 
   Do[allData = Import[fullPath, {"Data", sheet}]; colsWithData = {}; 
    If[Length[allData] >= rowStart, 
     sheetData = allData[[rowStart ;;]]; 
     Do[If[AnyTrue[Flatten[sheetData[[All, col]]], NumericQ], 
       AppendTo[colsWithData, col]], {col, 1, 
       Length[First[sheetData]]}];]; 
    AppendTo[result, Prepend[colsWithData, sheet]], {sheet, sheets}]; 
   result];

SeeTable[options___] := 
  Module[{params, filePath, sheetSpecs, axes, tick, plotLabel, 
    exportSpec, position, imagePadding, rowStart = RowStart, fullPath,
     sheets, allProcessedData = {}, processedData, nCurves, colors, 
    plotData, legendLabels, xlabel, ylabel, xLogQ = False, 
    yLogQ = False, scalingFunctions, legendScaledPos, plot, 
    exportPath, exportName, exportFullPath, hasLegend}, 
   params = <|options|>; 
   If[! KeyExistsQ[params, "path"], Return[$Failed]]; 
   filePath = params["path"]; sheetSpecs = filePath[[2]]; 
   If[! ListQ[First@sheetSpecs], sheetSpecs = {sheetSpecs}]; 
   axes = Lookup[params, "axes", {"x", "y"}]; xlabel = axes[[1]]; 
   ylabel = axes[[2]]; tick = Lookup[params, "tick", None]; 
   Switch[tick, "loglog", xLogQ = True; yLogQ = True, "log-", 
    xLogQ = True, "-log", yLogQ = True, _, Null]; 
   plotLabel = Lookup[params, "label", None]; 
   If[plotLabel === None, plotLabel = Lookup[params, "name", None]]; 
   exportSpec = Lookup[params, "export", None]; 
   position = Lookup[params, "position", Right]; 
   imagePadding = Lookup[params, "imagepadding", defaultImagePadding];
    fullPath = filePath[[1]]; 
   fullPath = 
    If[StringStartsQ[fullPath, "."] || StringStartsQ[fullPath, "\\"] ||
       StringStartsQ[fullPath, "/"], 
     FileNameJoin[{NotebookDirectory[], fullPath}], fullPath]; 
   If[! FileExistsQ[fullPath], Return[$Failed]]; 
   sheets = Import[fullPath, "Sheets"]; 
   Do[Module[{sheetName, xcol = 1, yspecs = {}, allData, sheetData}, 
     sheetName = spec[[1]]; 
     Do[If[Head[rule] === Rule, 
       Switch[rule[[1]], "xdata", xcol = rule[[2]], "ydata", 
        yspecs = rule[[2]], _, Null]], {rule, Rest@spec}]; 
     If[yspecs === {}, Continue[]]; 
     allData = Import[fullPath, {"Data", sheetName}]; 
     If[Length[allData] < rowStart, Continue[]]; 
     sheetData = allData[[rowStart ;;]]; 
     If[! ListQ[First@yspecs], yspecs = {yspecs}]; 
     Do[Module[{ycol, legend, pairs}, {ycol, legend} = yspec; 
       pairs = Transpose@{Flatten[sheetData[[All, xcol]]], 
          Flatten[sheetData[[All, ycol]]]}; 
       pairs = Select[pairs, 
         NumericQ[#[[1]]] && 
           NumericQ[#[[
             2]]] && (! xLogQ || #[[1]] > 0) && (! yLogQ || #[[2]] > 
              0) &]; If[pairs =!= {}, 
        AppendTo[allProcessedData, {pairs, legend}]];], {yspec, 
       yspecs}];], {spec, sheetSpecs}]; 
   If[allProcessedData === {}, Return[$Failed]]; 
   processedData = allProcessedData; nCurves = Length[processedData]; 
   colors = 
    If[nCurves <= 10, Take[ColorData[97, "ColorList"], nCurves], 
     ColorData["Rainbow"] /@ Subdivide[0, 1, nCurves]]; 
   plotData = processedData[[All, 1]]; 
   legendLabels = processedData[[All, 2]]; 
   scalingFunctions = 
    Which[xLogQ && yLogQ, {"Log10", "Log10"}, 
     xLogQ && ! yLogQ, {"Log10", None}, ! xLogQ && yLogQ, {None, 
      "Log10"}, True, Automatic]; 
   legendScaledPos = 
    Which[position === None, Right, 
     ListQ[position] && Length[position] == 2, Scaled[position], True,
      position]; 
   hasLegend = (position =!= 
       None) && (legendLabels =!= {}) && (legendLabels =!= {{}}); 
   plot = ListPlot[plotData, PlotStyle -> colors, 
     PlotMarkers -> {Automatic, Small}, Joined -> False, 
     Frame -> True, 
     FrameLabel -> 
      Map[Style[#, defaultFrameLabelStyle] &, {xlabel, ylabel}], 
     FrameTicksStyle -> defaultFrameStyle, 
     ScalingFunctions -> scalingFunctions, 
     PlotLabel -> 
      If[plotLabel =!= None, Style[plotLabel, defaultPlotLabelStyle], 
       None], ImageSize -> defaultImageSize, PlotRange -> Automatic, 
     GridLines -> None, 
     PlotLegends -> 
      If[hasLegend, 
       Placed[PointLegend[colors, legendLabels, 
         LegendLayout -> "Column", LabelStyle -> defaultLegendStyle], 
        legendScaledPos], None], BaseStyle -> defaultAxesStyle, 
     FrameStyle -> defaultFrameStyle, ImagePadding -> imagePadding]; 
   If[ListQ@exportSpec, {exportPath, exportName} = 
     Which[Length[exportSpec] >= 2, {exportSpec[[1]], 
       exportSpec[[2]]}]; 
    exportFullPath = FileNameJoin[{exportPath, exportName <> ".png"}];
     If[! DirectoryExistsQ@DirectoryName@exportFullPath, 
     CreateDirectory[DirectoryName@exportFullPath, 
      CreateIntermediateDirectories -> True]]; 
    Export[exportFullPath, plot, "PNG", ImageResolution -> 300, 
     Background -> White]; SystemOpen[exportFullPath];]; plot];

CompareResult[exprs_, 
   plotfor : {x_Symbol, xmin_?NumericQ, xmax_?NumericQ}, 
   "params" -> paramsSpec_, "change" -> changeSpec_, 
   "path" -> pathSpec_, opts : OptionsPattern[]] := 
  Module[{exprList, pathList, n, theoryRules, displayNames, 
    paramSymbols, displayAssoc, mentionedParams, changeRules, 
    expandedRules, theoryLegends, legendPos, axesList, labelList, 
    exportOpt, tickOption, theoryStyleList, theoryPlots, dataRaw, 
    dataPlots, dataLegends, dataLegendColors, combinedPlots, 
    allLegends, legendPosForData, changeSpecNorm}, 
   legendPos = OptionValue["position"]; 
   legendPosForData = If[legendPos === None, Right, legendPos]; 
   exportOpt = OptionValue["export"]; 
   tickOption = OptionValue["tick"]; 
   exprList = If[ListQ[exprs], exprs, {exprs}]; n = Length[exprList]; 
   pathList = 
    If[ListQ[pathSpec] && ListQ[First[pathSpec]], 
     pathSpec, {pathSpec}]; 
   If[Length[pathList] != n, 
    Return[$Failed]]; {theoryRules, displayNames, paramSymbols} = 
    processBaseParamsWithNames[paramsSpec]; 
   If[theoryRules === $Failed, Return[$Failed]]; 
   displayAssoc = AssociationThread[paramSymbols, displayNames]; 
   changeSpecNorm = normalizeChangeSpec[changeSpec]; 
   mentionedParams = getMentionedParams[changeSpecNorm]; 
   changeRules = 
    processChangeSpecWithNames[changeSpecNorm, theoryRules, 
     displayAssoc]; If[changeRules === $Failed, Return[$Failed]]; 
   expandedRules = 
    generateParameterCombinationsSimple[theoryRules, changeRules]; 
   If[expandedRules === $Failed, Return[$Failed]]; 
   theoryLegends = 
    generateLegendsWithNames[expandedRules, mentionedParams, 
     displayAssoc]; 
   theoryStyleList = 
    If[OptionValue[PlotStyle] === Automatic, 
     Table[ColorData[97][k], {k, Length[expandedRules]}], 
     OptionValue[PlotStyle]]; 
   axesList = normalizeAxesSpec[OptionValue["axes"], x, exprList]; 
   If[axesList === $Failed, Return[$Failed]]; 
   labelList = normalizeLabelSpec[OptionValue["label"], n]; 
   If[labelList === $Failed, Return[$Failed]]; 
   theoryPlots = 
    Table[Module[{theoryCurves}, 
      theoryCurves = 
       Table[exprList[[i]] /. rules, {rules, expandedRules}]; 
      Plot[theoryCurves, plotfor, PlotStyle -> theoryStyleList, 
       BaseStyle -> defaultAxesStyle, PlotRange -> Automatic, 
       Frame -> OptionValue[Frame], 
       FrameLabel -> 
        If[OptionValue[Frame] && axesList[[i]] =!= None, 
         Map[Style[#, defaultFrameLabelStyle] &, axesList[[i]]], 
         None], FrameTicksStyle -> defaultFrameStyle, 
       AxesLabel -> 
        If[! OptionValue[Frame] && axesList[[i]] =!= None, 
         Map[Style[#, defaultFrameLabelStyle] &, axesList[[i]]], 
         None], GridLines -> None, 
       ImageSize -> OptionValue[ImageSize], 
       LabelStyle -> OptionValue["LabelStyle"], 
       PlotLabel -> 
        If[labelList[[i]] =!= None, 
         Style[labelList[[i]], defaultPlotLabelStyle], None], 
       Background -> White, FrameStyle -> defaultFrameStyle, 
       PlotLegends -> None, 
       ImagePadding -> OptionValue[ImagePadding]]], {i, n}]; 
   dataRaw = 
    Table[SeeTable["path" -> pathList[[i]], 
      "axes" -> (axesList[[i]] /. None -> {"", ""}), 
      "tick" -> tickOption, "label" -> labelList[[i]], 
      "position" -> legendPosForData, "export" -> None], {i, n}]; 
   If[MemberQ[dataRaw, $Failed], Return[$Failed]]; 
   dataLegends = 
    Table[FirstCase[dataRaw[[i]], 
      PointLegend[_, labels_, ___] :> labels, {}, Infinity], {i, n}]; 
   dataLegendColors = 
    Table[FirstCase[dataRaw[[i]], 
      PointLegend[cols_, _, ___] :> cols, {}, Infinity], {i, n}]; 
   dataPlots = dataRaw /. Legended[g_, __] :> g; 
   allLegends = 
    Table[Module[{hasLegend = False, expLegend = None, 
       theoryLegend = None}, 
      If[dataLegends[[i]] =!= {} && dataLegendColors[[i]] =!= {}, 
       hasLegend = True; 
       expLegend = 
        PointLegend[dataLegendColors[[i]], dataLegends[[i]], 
         LegendLayout -> "Column", LegendMarkerSize -> 15, 
         LegendMarkers -> {Automatic, 15}, 
         LabelStyle -> defaultLegendStyle];]; 
      If[theoryLegends =!= {} && theoryLegends =!= {{}} && 
        theoryLegends =!= {""}, hasLegend = True; 
       theoryLegend =
