\chapter{主方程的应用：所有具体问题都只是构造同一个 \texorpdfstring{$\beta$}{beta}}
\label{chap:master-applications}
\label{chap:direct-applications}

第~\ref{chap:pinem-core} 章已经把相互作用区内的电子问题全部压缩成
\begin{equation}
\boxed{
\beta[\bm E_\omega]
=-\frac{q_{\rm e}}{2\hbar\omega}
\int_{-\infty}^{+\infty}\dd s\,
\hat{\bm v}_0\cdot\bm E_\omega(\bm R_\perp+s\hat{\bm v}_0)
\ee^{-\ii\kappa s},
\qquad
\kappa\equiv\frac{\omega}{v_0}}
\label{eq:application-beta-master}
\end{equation}
以及
\begin{equation}
\boxed{
 c_n=\ee^{\ii n\arg(-\beta)}J_n(2|\beta|),
 \qquad
 P_n=J_n^2(2|\beta|).}
\label{eq:application-response-master}
\end{equation}
所以本章所有例子都按同一顺序阅读：先写具体的 \(E_{\omega,\parallel}\)，算出 \(\beta\)，然后停止。Bessel 响应不再重推。

\section{纵向场型只是同一个 Fourier 取样器的不同输入}

束轴取为 \(+z\)。定义
\begin{equation}
\widetilde E_z(k;\bm R_\perp)
\equiv
\int_{-\infty}^{+\infty}\dd z\,
E_{\omega,z}(\bm R_\perp,z)\ee^{-\ii kz},
\end{equation}
则式~\eqref{eq:application-beta-master} 立即化成
\begin{equation}
\boxed{
\beta(\bm R_\perp)
=-\frac{q_{\rm e}}{2\hbar\omega}
\widetilde E_z\!\left(\frac{\omega}{v_0};\bm R_\perp\right).}
\label{eq:beta-fourier-sample}
\end{equation}
这是一切单频应用最短的数学结构：电子只读取纵向空间谱在 \(k_z=\omega/v_0\) 处的复振幅。

例如
\begin{equation}
E_{\omega,z}(z)=E_0\ee^{-z^2/(2L^2)}
\end{equation}
给出
\begin{equation}
\boxed{
\beta
=-\frac{q_{\rm e}E_0}{2\hbar\omega}\sqrt{2\pi}L
\exp\!\left[-\frac12\left(\frac{\omega L}{v_0}\right)^2\right].}
\label{eq:beta-gaussian-field}
\end{equation}
场越平滑，所需高空间频率越少，耦合越快被指数压低。反过来，局域得无限尖锐也不意味着耦合无限增大。若
\begin{equation}
E_{\omega,z}(z)=E_0\ee^{-|z|/\ell},
\end{equation}
则
\begin{equation}
\boxed{
\beta
=-\frac{q_{\rm e}E_0}{\hbar\omega}
\frac{\ell}{1+(\kappa\ell)^2}.}
\label{eq:exponential-localization-beta}
\end{equation}
在固定峰值场 \(E_0\) 下，对 \(\ell\) 求极值可得
\begin{equation}
\boxed{\ell_{\rm opt}=\frac{1}{\kappa}=\frac{v_0}{\omega}.}
\label{eq:optimal-longitudinal-localization}
\end{equation}
太宽时缺少 \(k_z=\kappa\) 的空间谱，太窄时有效作用长度又太短；最优尺度正好是电子在一个光学弧度时间 \(1/\omega\) 内走过的距离。

若长度 \(L\) 内存在近似行波
\begin{equation}
E_{\omega,z}(z)=E_0\ee^{\ii qz},
\qquad |z|<L/2,
\end{equation}
则
\begin{equation}
\boxed{
\beta
=-\frac{q_{\rm e}E_0L}{2\hbar\omega}
\frac{\sin[(q-\kappa)L/2]}{(q-\kappa)L/2}.}
\label{eq:finite-length-sinc-matching}
\end{equation}
无限长度只保留 \(q=\kappa\)；有限长度把严格相位匹配展宽成 sinc 线型。

一个更接近表面模的输入是
\begin{equation}
E_{\omega,z}(b,z)=E_0\ee^{-\chi b}\ee^{\ii qz},
\qquad |z|<L/2,
\end{equation}
其中 \(b\) 为电子到界面的冲量参数。直接代入得到
\begin{equation}
\boxed{
\beta(b)
=-\frac{q_{\rm e}E_0L}{2\hbar\omega}
\ee^{-\chi b}
\frac{\sin[(q-\kappa)L/2]}{(q-\kappa)L/2}.}
\label{eq:evanescent-traveling-beta}
\end{equation}
同一个式子同时展示两件独立的事：\(\ee^{-\chi b}\) 决定横向接近结构时的增强，sinc 因子决定纵向相位匹配。二者不应混成一个“耦合长度”。若固定结构模波数 \(q\) 而扫描电子速度，相位匹配中心为
\begin{equation}
\boxed{v_*=\frac{\omega}{q}.}
\label{eq:velocity-phase-match-center}
\end{equation}
在 \(v_0=v_*+\delta v\) 附近，
\begin{equation}
q-\frac{\omega}{v_0}
\simeq\frac{\omega}{v_*^2}\delta v.
\end{equation}
因此有限长度 sinc 的第一个零点对应
\begin{equation}
\boxed{
|\delta v|_{\rm zero}
\simeq\frac{2\pi v_*^2}{\omega L}.}
\label{eq:velocity-phase-match-width}
\end{equation}
作用区越长，电子速度扫描对模式纵向波数的选择越尖锐。

均匀宿主中的传播分量满足
\begin{equation}
k_\parallel^2+k_\perp^2=\left(\frac{n_{\rm h}\omega}{c}\right)^2.
\end{equation}
而电子要求 \(k_\parallel=\omega/v_0\)。若 \(v_0<c/n_{\rm h}\)，则所需点位于传播光锥之外，必须依赖倏逝波、波导模、表面模或结构散射提供高空间频率；对应的横向衰减常数为
\begin{equation}
\boxed{
\chi_\perp
=\sqrt{\left(\frac{\omega}{v_0}\right)^2
-\left(\frac{n_{\rm h}\omega}{c}\right)^2}.}
\label{eq:evanescent-decay-from-kinematics}
\end{equation}
这不是另一条 PINEM 定律，而只是式~\eqref{eq:beta-fourier-sample} 与 Maxwell 色散同时成立后的几何结果。

\section{利用线性性设计复耦合，而不是重推电子动力学}

式~\eqref{eq:application-beta-master} 对电场严格线性。若
\begin{equation}
\bm E_\omega=\sum_a u_a\bm E_\omega^{(a)},
\end{equation}
则
\begin{equation}
\boxed{\beta=\sum_a u_a\beta_a.}
\label{eq:beta-linearity-superposition}
\end{equation}
因此多区干涉、偏振控制、阵列增强和参考场测相位，本质上都是复平面里的矢量相加。

两个相干作用区给出
\begin{equation}
\beta_{\rm tot}=\beta_1+\ee^{\ii\Phi}\beta_2.
\label{eq:two-zone-beta}
\end{equation}
若 \(\beta_1=\beta_2=\beta_0\)，则
\begin{equation}
\boxed{
P_n(\Phi)
=J_n^2\!\left(4|\beta_0|\left|\cos\frac{\Phi}{2}\right|\right).}
\label{eq:two-zone-pinem-interference}
\end{equation}
这个结果成立的前提是两个作用区之间的自由传播尚未积累不可忽略的边带二次色散相位；若漂移足够长，应先按第~\ref{chap:downstream-observables} 章传播各个 \(c_n\)，再施加第二次相互作用。

对等间距的 \(N\) 个相同单元，若第 \(j\) 个单元相对电子多出相位 \((q-\kappa)jd\)，则
\begin{align}
\beta_N
&=\beta_1\sum_{j=0}^{N-1}\ee^{\ii(q-\kappa)jd}\\
&=\boxed{
\beta_1\ee^{\ii(N-1)\Delta kd/2}
\frac{\sin(N\Delta kd/2)}{\sin(\Delta kd/2)}},
\qquad \Delta k=q-\kappa.
\label{eq:array-beta-sinc}
\end{align}
阵列、准相位匹配和有限单元数的主瓣宽度因此都只是 \(\beta\) 的结构因子。

偏振也完全相同。若入射场写成
\begin{equation}
\bm E_\omega=u_x\bm E_x+u_y\bm E_y,
\end{equation}
那么
\begin{equation}
\boxed{\beta(u_x,u_y)=u_x\beta_x+u_y\beta_y.}
\label{eq:polarization-beta}
\end{equation}
实验扫描偏振，本质上是在复平面中扫描 \(\beta\)。

\section{横向相位结构直接乘到边带相位上}

如果轨迹冻结近似下
\begin{equation}
\beta(\bm R_\perp)=|\beta(\bm R_\perp)|\ee^{\ii\Phi(\bm R_\perp)},
\end{equation}
则第 \(n\) 个边带振幅自动获得
\begin{equation}
\boxed{
c_n(\bm R_\perp)
=J_n\!\left(2|\beta|\right)
\ee^{\ii n[\Phi(\bm R_\perp)+\pi]},}
\label{eq:general-transverse-phase-multiplication}
\end{equation}
其中固定的 \(\pi\) 只来自 \(\arg(-\beta)\) 的约定。也就是说，局域光学相位被第 \(n\) 个能量通道放大为 \(n\Phi\)。

轴对称涡旋场是这一结论的直接特例。若
\begin{equation}
\beta(\rho,\varphi)=\beta_\ell(\rho)\ee^{\ii\ell\varphi},
\end{equation}
则
\begin{equation}
c_n\propto\ee^{\ii n\ell\varphi}.
\end{equation}
若入射电子横向态含 \(\ee^{\ii m\varphi}\)，并且轴对称性使其他角动量混合可以忽略，则
\begin{equation}
\boxed{m_{\rm out}=m+n\ell,
\qquad \Delta L_z=n\ell\hbar.}
\label{eq:oam-selection-rule}
\end{equation}
同时
\begin{equation}
P_n(\rho)=J_n^2\!\left(2|\beta_\ell(\rho)|\right)
\end{equation}
不含 \(\varphi\)。因此普通能谱只见到 \(|\beta|\)，看不到这类横向相位；相位必须通过模式分析或参考耦合读出。

\section{有限脉冲和多频场只是让输入从一个复数变成一组复数}

准单色脉冲写成
\begin{equation}
\bm E^{(+)}(\bm r,t)
=\bm E_\omega(\bm r)a_{\rm L}(t-\tau)\ee^{-\ii\omega t},
\end{equation}
并要求
\begin{equation}
\frac{|\partial_ta_{\rm L}|}{\omega|a_{\rm L}|}\ll1
\end{equation}
以及脉冲带宽内空间模式变化足够小。对穿过参考面时刻为 \(t_{\rm e}\) 的电子，沿轨迹直接代入包络：
\begin{equation}
\boxed{
\beta_{\rm p}(t_{\rm e},\tau)
=-\frac{q_{\rm e}}{2\hbar\omega}
\int\dd s\,
E_{\omega,\parallel}(s)
 a_{\rm L}\!\left(t_{\rm e}+\frac{s}{v_0}-\tau\right)
\ee^{-\ii\kappa s}.}
\label{eq:pulse-weighted-beta}
\end{equation}
然后仍然只有
\begin{equation}
\boxed{
 c_n(t_{\rm e},\tau)
 =\ee^{\ii n\arg[-\beta_{\rm p}(t_{\rm e},\tau)]}
 J_n\!\left(2|\beta_{\rm p}(t_{\rm e},\tau)|\right),}
\label{eq:pulse-local-sideband-amplitude}
\end{equation}
以及
\begin{equation}
P_n(t_{\rm e},\tau)=J_n^2\!\left(2|\beta_{\rm p}(t_{\rm e},\tau)|\right).
\end{equation}
若近场穿越时间 \(T_{\rm tr}=L_{\rm nf}/v_0\) 满足 \(T_{\rm tr}\ll\sigma_{\rm L}\)，则
\begin{equation}
\boxed{
\beta_{\rm p}(t_{\rm e},\tau)
\simeq\beta\,a_{\rm L}(t_{\rm e}-\tau).}
\label{eq:beta-local-def}
\end{equation}
这就是所有短穿越有限脉冲公式的母式。

若场含多个相干离散频率 \(\omega_j\)，每个频率先独立计算
\begin{equation}
\beta_j
=-\frac{q_{\rm e}}{2\hbar\omega_j}
\int\dd s\,E_{\omega_j,\parallel}(s)
\ee^{-\ii\omega_js/v_0}.
\end{equation}
由于一阶 eikonal 相位可加，输出因子分解为
\begin{equation}
\frac{f_{\rm out}}{f_{\rm in}}
=\prod_j
\exp\!\left[-\beta_j\ee^{-\ii\omega_jt_c}
+\beta_j^*\ee^{+\ii\omega_jt_c}\right].
\label{eq:multifrequency-product-main}
\end{equation}
于是通道由整数向量 \(\bm n=(n_1,n_2,\ldots)\) 标记，并具有
\begin{equation}
\boxed{
\Delta\mathcal E_{\bm n}
=\sum_jn_j\hbar\omega_j.}
\label{eq:multifrequency-energy-main}
\end{equation}
若不同整数路径落到同一总能量，它们必须在振幅层相加。完整展开留在附录~\ref{app:multifrequency-chapter}；这里要强调的仍然只是“每个频率各算一个 \(\beta_j\)”。

\section{材料和几何只负责先产生电场：Mie 与 Rayleigh 也只是输入生成器}

当 \(\bm E_\omega\) 不是预先给定，而是由纳米结构决定时，电子端没有任何变化：
\begin{equation}
\boxed{
\text{材料与几何}
\longrightarrow \bm E_\omega
\longrightarrow \beta[\bm E_\omega]
\longrightarrow \{c_n,P_n\}.}
\label{eq:maxwell-interface-short}
\end{equation}
任意线性结构都可以用 Green/T 算符求场；球形结构可以用 VSH/VSWF 和 Mie 系数精确求场。这些 Maxwell 推导集中放在附录~\ref{chap:maxwell-general}，因为它们只解决第一支箭头。

对单球，精确结果最后总可以压成
\begin{equation}
\boxed{
\beta_{\rm sc}
=-\frac{q_{\rm e}}{2\hbar\omega}
\sum_{\ell m_\Omega}
\left[
 b_\ell p_{\ell m_\Omega}\mathcal I_{\ell m_\Omega}^{(M)}
+a_\ell q_{\ell m_\Omega}\mathcal I_{\ell m_\Omega}^{(E)}
\right],}
\label{eq:mie-beta-short}
\end{equation}
其中 \(a_\ell,b_\ell\) 只含球材料和尺寸，\(p_{\ell m_\Omega},q_{\ell m_\Omega}\) 只含入射场，轨迹核 \(\mathcal I^{(M/E)}\) 只含电子几何。三类因素在线性层面彼此分离。

Rayleigh 极限甚至可以直接给出一个显式的 \(\beta\)。设均匀宿主介电常数为 \(\varepsilon_0\varepsilon_{\rm h}\)，亚波长粒子等效为沿电子方向的准静态偶极 \(\bm p=p_z\hat{\bm z}\)，电子沿 \(z\) 方向以冲量参数 \(b\) 掠过。偶极纵向场为
\begin{equation}
E_z(b,z)
=\frac{p_z}{4\pi\varepsilon_0\varepsilon_{\rm h}}
\frac{2z^2-b^2}{(b^2+z^2)^{5/2}}.
\end{equation}
利用
\begin{equation}
\int_{-\infty}^{+\infty}\frac{\ee^{-\ii\kappa z}}{\sqrt{b^2+z^2}}\dd z
=2K_0(\kappa b),
\end{equation}
并注意 \(E_z=(p_z/4\pi\varepsilon_0\varepsilon_{\rm h})\partial_z^2(b^2+z^2)^{-1/2}\)，得到
\begin{equation}
\int E_z\ee^{-\ii\kappa z}\dd z
=-\frac{p_z\kappa^2}{2\pi\varepsilon_0\varepsilon_{\rm h}}K_0(\kappa b).
\end{equation}
因此
\begin{equation}
\boxed{
\beta_{\rm dip}(b)
=\frac{q_{\rm e}p_z}{4\pi\varepsilon_0\varepsilon_{\rm h}\hbar\omega}
\left(\frac{\omega}{v_0}\right)^2
K_0\!\left(\frac{\omega b}{v_0}\right).}
\label{eq:dipole-beta-K0}
\end{equation}
若球半径为 \(a\)，在非磁、准静态条件下
\begin{equation}
p_z=\alpha_0E_{{\rm inc},z},
\qquad
\alpha_0=4\pi\varepsilon_0\varepsilon_{\rm h}a^3
\frac{\varepsilon_{\rm p}-\varepsilon_{\rm h}}
{\varepsilon_{\rm p}+2\varepsilon_{\rm h}}.
\end{equation}
于是材料共振只进入 \(\alpha_0\)，冲量参数和电子速度只进入 \(K_0(\omega b/v_0)\)。这正是“材料响应、几何衰减和电子相位匹配彼此分离”的一个最短例子。

本章到这里结束。所有场景都没有改变式~\eqref{eq:application-response-master}；它们只是给出了不同的 \(\beta\) 或 \(\{\beta_j\}\)。下一章只问：得到这些边带以后，能从中读出什么。
