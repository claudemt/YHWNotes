\chapter{特征标与张量积}
\label{chap:characters-projections}

逐个寻找相似变换来分块往往比原问题还难。迹在换基下不变，却仍保留足够多的群信息；把每个群元的表示矩阵取迹，得到的特征标只在共轭类上变化。

Schur 引理会把群平均变成特征标正交关系。由此既能用一个内积读出不可约表示的重数，也能构造投影算符，把具体轨道、位移或函数投到指定对称类型中。特征标因而不是矩阵的廉价摘要，而是实际完成分解的计算工具。

\section{特征标理论}
\label{sec:character-theory}

大正交关系处理的是选定基中的矩阵元。若只想判断一个表示包含哪些不可约类型，
可以把矩阵元对角求和：换基 $D'(g)=S^{-1}D(g)S$ 不改变迹，
因此 $g\mapsto\Tr D(g)$ 只依赖表示的等价类。
下面将用大正交关系证明这些迹函数彼此正交，再通过内积求出未知表示的重数。

\begin{definition}[特征标]
设 $\rho:G\to\operatorname{GL}(V)$ 为有限维表示，$D(g)$ 为任意一组基下的表示矩阵。定义函数
\[
\chi_\rho:G\to\CC,
\qquad
\chi_\rho(g)=\Tr D(g).
\]
它称为表示 $\rho$ 的\textbf{特征标}。
\label{def:character}
\end{definition}

\begin{proposition}[特征标是类函数]
对任意表示 $\rho$ 与任意 $x,g\in G$，有
\[
\chi_\rho(xgx^{-1})=\chi_\rho(g).
\]
因此特征标在共轭类上常值，称为\textbf{类函数}。
\label{prop:character-class-function}

\end{proposition}

\begin{proof}
由表示的同态性质，
\[
D(xgx^{-1})=D(x)D(g)D(x)^{-1}.
\]
取迹并利用迹对相似变换不变，得到
\[
\chi_\rho(xgx^{-1})=\Tr(D(x)D(g)D(x)^{-1})=\Tr(D(g))=\chi_\rho(g).
\]
\end{proof}

这一定理非常重要，因为它把“表示理论”与\chapref{chap:group-basics}的“共轭类结构”直接连接了起来。以后我们做特征标表时，行由不可约表示标记，列由共轭类标记，根本原因就在于特征标只依赖于共轭类。

对类函数空间引入内积
\[
\langle f_1,f_2\rangle
=\frac1{|G|}\sum_{g\in G} \overline{f_1(g)}f_2(g).
\]
在这个内积下，不可约特征标会构成一组标准正交基。

\begin{theorem}[不可约特征标的正交关系]
设 $\chi^{(\alpha)}$ 与 $\chi^{(\beta)}$ 是有限群 $G$ 的两个不可约复表示的特征标，则
\[
\langle \chi^{(\alpha)},\chi^{(\beta)}\rangle
=\frac1{|G|}\sum_{g\in G}\overline{\chi^{(\alpha)}(g)}\chi^{(\beta)}(g)
=\delta_{\alpha\beta}.
\]
\label{thm:character-orthogonality}

\end{theorem}

\begin{proof}
把特征标写成矩阵元对角和，得
\[
\chi^{(\alpha)}(g)=\sum_i D^{(\alpha)}_{ii}(g),
\qquad
\chi^{(\beta)}(g)=\sum_k D^{(\beta)}_{kk}(g).
\]
于是
\begin{align*}
\frac1{|G|}\sum_{g\in G}\overline{\chi^{(\alpha)}(g)}\chi^{(\beta)}(g)
&=\frac1{|G|}\sum_{g\in G}\sum_{i,k}
\overline{D^{(\alpha)}_{ii}(g)}D^{(\beta)}_{kk}(g)\\
&=\frac1{|G|}\sum_{i,k}
\sum_{g\in G}\overline{D^{(\alpha)}_{ii}(g)}D^{(\beta)}_{kk}(g).
\end{align*}
应用 \Cref{thm:great-orthogonality}，内层求和变为
\[
\frac{|G|}{d_\alpha}\delta_{\alpha\beta}\delta_{ik}.
\]
再对 $i,k$ 求和，得到
\[
\frac1{|G|}\cdot \frac{|G|}{d_\alpha}\delta_{\alpha\beta}\sum_i 1
=\delta_{\alpha\beta}.
\]
故结论成立。
\end{proof}

\begin{corollary}[重数公式]
设表示 $\rho$ 的特征标为 $\chi$，其不可约分解为
\[
\rho\simeq \bigoplus_\alpha m_\alpha \rho^{(\alpha)}.
\]
则每个重数都满足
\[
m_\alpha=\langle \chi^{(\alpha)},\chi\rangle
=\frac1{|G|}\sum_{g\in G}\overline{\chi^{(\alpha)}(g)}\chi(g).
\]
\label{cor:multiplicity-formula}

\end{corollary}

\begin{proof}
由于特征标对直接和可加，
\[
\chi=\sum_\beta m_\beta \chi^{(\beta)}.
\]
两边与 $\chi^{(\alpha)}$ 取内积，利用正交关系即得
\[
\langle \chi^{(\alpha)},\chi\rangle
=\sum_\beta m_\beta\langle \chi^{(\alpha)},\chi^{(\beta)}\rangle
=m_\alpha.
\]
\end{proof}

\begin{proposition}[投影算符的显式公式]
\label{prop:projection-operator}
设 $V=\bigoplus_\alpha m_\alpha V_\alpha$ 是 $G$ 的表示 $V$ 的不可约分解。对每个不可约表示 $\alpha$，投影到 $\alpha$-同类分量 $V[\alpha]\cong m_\alpha V_\alpha$ 的 $G$-等变投影算符为
\begin{equation}
P_\alpha=\frac{d_\alpha}{|G|}\sum_{g\in G}\overline{\chi_\alpha(g)}\,\rho(g).
\label{eq:projection-operator}
\end{equation}
它满足
\[
P_\alpha^2=P_\alpha,\qquad P_\alpha P_\beta=0\ (\alpha\neq\beta),\qquad\sum_\alpha P_\alpha=I_V.
\]

\end{proposition}

\begin{proof}
对任意 $h\in G$，共轭只是在求和中重排群元，而特征标在共轭类上常值，因此
\[
\rho(h)P_\alpha\rho(h)^{-1}=\frac{d_\alpha}{|G|}\sum_g\overline{\chi_\alpha(g)}\rho(hgh^{-1})
=\frac{d_\alpha}{|G|}\sum_{g'}\overline{\chi_\alpha(g')}\rho(g')
=P_\alpha,
\]
故 $P_\alpha$ 与表示的作用交换。

将它限制到不可约分量 $V_\beta$ 上。Schur 引理给出
$P_\alpha|_{V_\beta}=\lambda_\beta I_{V_\beta}$，而取迹可得
\[
\Tr_{V_\beta}P_\alpha=\frac{d_\alpha}{|G|}\sum_g\overline{\chi_\alpha(g)}\chi_\beta(g)=d_\alpha\langle\chi_\alpha,\chi_\beta\rangle=d_\alpha\delta_{\alpha\beta}.
\]
于是 $\lambda_\beta d_\beta=d_\alpha\delta_{\alpha\beta}$，即
$\lambda_\beta=\delta_{\alpha\beta}$。所以 $P_\alpha$ 在 $V[\alpha]$ 上为恒等、在其余同类分量上为零；幂等性、相互正交性与 $\sum_\alpha P_\alpha=I$ 随即成立。
\end{proof}

\begin{example}[投影算符在对称性适配线性组合构造中的应用]
\label{ex:projection-salc}
设三个等价位置上的轨道 $\phi_1,\phi_2,\phi_3$ 正交归一，$C_{3v}\cong D_3$ 置换这三个轨道。
用\eqnrefs{eq:projection-operator} 求 $A_1$ 与 $E$ 的对称性适配线性组合，并判断 $A_2$ 是否出现。

\textbf{解。} 按共轭类 $(e,3\sigma_v,2C_3)$ 的顺序，置换表示特征标为 $(3,1,0)$。
将 $A_1=(1,1,1)$、$A_2=(1,-1,1)$、$E=(2,0,-1)$ 代入重数公式，得到
\[
m_{A_1}=\frac{3+3}6=1,\qquad
m_{A_2}=\frac{3-3}6=0,\qquad
m_E=\frac{2\cdot3}6=1.
\]
这里二维 $E$ 是三角形平面上的转动与反射表示，其矩阵已在前章构造。
对平凡表示，群平均投影为
\[
P_{A_1}=\frac16\bigl(I+U(r)+U(r^2)+U(s)+U(sr)+U(sr^2)\bigr),
\]
每个目标轨道在六项中出现两次，所以
\[
P_{A_1}\phi_1=\frac13(\phi_1+\phi_2+\phi_3),\qquad
\psi_{A_1}=\frac{\phi_1+\phi_2+\phi_3}{\sqrt3}.
\]
对二维表示，$\chi_E$ 在反射上为零，因此
\[
P_E=\frac13\bigl(2I-U(r)-U(r^2)\bigr),\qquad
P_E\phi_1=\frac13(2\phi_1-\phi_2-\phi_3).
\]
这个向量的范数为 $\sqrt{2/3}$。归一化后，再在分量和为零的空间中取一个正交向量，得到
\[
\psi_{E,1}=\frac{2\phi_1-\phi_2-\phi_3}{\sqrt6},\qquad
\psi_{E,2}=\frac{\phi_2-\phi_3}{\sqrt2}.
\]
两者都被 $P_E$ 保持，且与 $\psi_{A_1}$ 正交，完整给出 $A_1\oplus E$ 的适配基。
若原子轨道存在重叠，最后一步应按实际重叠矩阵计算范数，不能直接沿用这里的归一化系数。
\end{example}


重数公式的使用方式与 Fourier 展开完全一致。若
\[
\chi=\sum_\beta m_\beta\chi^{(\beta)},
\]
不可约特征标就是一组正交基。要取出某一个系数 $m_\alpha$，只需与对应基向量做内积：
\[
m_\alpha=\langle\chi^{(\alpha)},\chi\rangle_G.
\]
计算时不必逐个群元求和，因为特征标在同一共轭类上相同。若共轭类为 $C_r$，任选代表元 $g_r\in C_r$，则
\[
m_\alpha=\frac1{|G|}\sum_r |C_r|\,\overline{\chi^{(\alpha)}(g_r)}\chi(g_r).
\]
这就是特征标表中为什么必须同时列出每个类的元素个数。

这个公式几乎是实际计算不可约分解的标准工具：只要知道某个表示的特征标，就可以不显式找不变子空间，而直接算出每个不可约表示出现多少次。对点群振动模式、轨道杂化与多重简并的分类，它尤其方便。

在进入下面的完备性证明以前，先把其中两个新空间翻译成人话。若 $G$ 有 $k$ 个共轭类 $C_1,\dots,C_k$，一个类函数 $f$ 在每个类上只能取一个共同值，因此它完全由 $k$ 个复数
\[
f(C_1),\dots,f(C_k)
\]
决定。反过来，任意指定这 $k$ 个数都得到一个类函数。所以类函数空间本身就是一个 $k$ 维向量空间：
\[
\dim\mathcal C(G)=k.
\]

同样的“一个共轭类只对应一个自由系数”也出现在群代数的中心。写一个一般群代数元素
\[
z=\sum_{g\in G}a_g g.
\]
它属于中心 $Z(\CC[G])$ 的条件是对任意 $h\in G$ 都有 $hzh^{-1}=z$。把左边展开：
\[
hzh^{-1}=\sum_{g\in G}a_g\,hgh^{-1}.
\]
比较群元基的系数可知，中心性等价于
\[
a_{hgh^{-1}}=a_g
\qquad\forall g,h\in G,
\]
也就是系数在每个共轭类上常值。因此若定义类和
\[
K_r=\sum_{g\in C_r}g,
\]
那么 $K_1,\dots,K_k$ 构成群代数中心的一组基。于是“类函数空间”和“群代数中心”虽然对象不同，却都由同一批共轭类坐标化。下面证明中把类函数 $f$ 包装成中心元 $z_f$，正是在利用这层一一对应。

\begin{proposition}[置换特征标与轨道数]
设有限群 $G$ 作用在有限集 $X$ 上，置换表示的特征标为 $\chi_{\mathrm{perm}}$。
则 $\chi_{\mathrm{perm}}(g)=|X^g|$，且其平凡表示重数等于 $|X/G|$。
\end{proposition}

\begin{proof}
置换矩阵的对角元为 $1$ 当且仅当相应基标签被固定，所以迹为 $|X^g|$。
由重数公式与\cref{thm:burnside}，
\[
\langle\chi_{\mathrm{triv}},\chi_{\mathrm{perm}}\rangle
=\frac1{|G|}\sum_g|X^g|=|X/G|.
\]
也可直接构造不变向量：每个轨道上基向量的和不变，不同轨道给出独立向量；
任何不变向量的系数在各轨道上必须相同，因此它们构成全部不变空间的一组基。
\end{proof}

\begin{theorem}[不可约特征标构成类函数空间的一组正交基]
有限群 $G$ 的全部不可约特征标构成类函数空间 $\mathcal C(G)$ 的一组正交基。因而，不等价不可约表示的个数等于 $G$ 的共轭类个数。
\label{thm:character-basis}

\end{theorem}

\begin{proof}
正交性已经由 \Cref{thm:character-orthogonality} 给出，所以不可约特征标线性无关。真正需要证明的是：若一个类函数与所有不可约特征标都正交，那么它只能是零函数。

设 $f\in\mathcal C(G)$ 满足
\[
\langle\chi^{(\alpha)},f\rangle=0
\qquad
\text{对每个不可约表示 }\alpha.
\]
在群代数 $\CC[G]$ 中构造
\begin{equation}
z_f=\sum_{g\in G}f(g^{-1})g.
\label{eq:class-function-central-element}
\end{equation}
由于 $f$ 在共轭类上常值，$g\mapsto f(g^{-1})$ 也是类函数，所以对任意 $h\in G$，
\begin{align*}
hz_fh^{-1}
&=\sum_{g\in G}f(g^{-1})hgh^{-1}\\
&=\sum_{k\in G}f(k^{-1})k=z_f,
\end{align*}
其中第二步令 $k=hgh^{-1}$。因此 $z_f$ 位于群代数中心。

令不可约表示 $D^{(\alpha)}$ 线性延拓到群代数。因为 $z_f$ 是中心元，$D^{(\alpha)}(z_f)$ 与所有 $D^{(\alpha)}(h)$ 对易；Schur 引理于是给出
\[
D^{(\alpha)}(z_f)=c_\alpha I_{d_\alpha}.
\]
取迹并令 $k=g^{-1}$，再使用不可约表示可酉化所给出的
$\chi^{(\alpha)}(k^{-1})=\overline{\chi^{(\alpha)}(k)}$，得到
\begin{align*}
d_\alpha c_\alpha
&=\sum_{g\in G}f(g^{-1})\chi^{(\alpha)}(g)\\
&=\sum_{k\in G}f(k)\chi^{(\alpha)}(k^{-1})\\
&=|G|\,\langle\chi^{(\alpha)},f\rangle=0.
\end{align*}
所以每个不可约表示都把 $z_f$ 送到零算子。

现在作用到左正则表示。由 Maschke 定理，正则表示可以分解成不可约表示的直和，因此 $z_f$ 在每个不可约直和分量上都作用为零，故左乘算子 $L(z_f)$ 整体为零。可是群代数的左正则表示是忠实的：若 $e$ 表示群单位元所对应的基向量，则
\[
L(z_f)e=z_f.
\]
因此 $L(z_f)=0$ 迫使 $z_f=0$。比较 \eqnrefs{eq:class-function-central-element} 中各群元的系数，得到
\[
f(g^{-1})=0,
\qquad\forall g\in G,
\]
也就是 $f=0$。

所以不可约特征标的正交补只有零向量，它们必张成整个有限维类函数空间 $\mathcal C(G)$。而类函数由每个共轭类上的常值唯一确定，因此
\[
\dim\mathcal C(G)=\#\{\text{$G$ 的共轭类}\}.
\]
于是不可约表示等价类数恰好等于共轭类数。
\end{proof}

这个证明还解释了为什么“共轭类数 = 不可约表示数”并不是一个孤立的计数巧合。类函数 $f$ 被包装成中心元 $z_f$；Schur 引理把中心元在每个不可约块上的作用压缩成一个标量；正则表示的忠实性再保证，如果这些标量全部消失，原来的中心元本身就必须消失。于是共轭类、群代数中心、不可约表示三种看似不同的结构在这里真正接到了一起。

\begin{corollary}[正则表示的分解与维数平方和公式]
设 $\rho_{\mathrm{reg}}$ 为有限群 $G$ 的左正则表示，$\rho^{(\alpha)}$ 取遍全部两两不等价的不可约表示，维数记为 $d_\alpha$。则
\[
\rho_{\mathrm{reg}}\simeq \bigoplus_\alpha d_\alpha\rho^{(\alpha)}.
\]
特别地，
\[
|G|=\sum_\alpha d_\alpha^2.
\]
\label{cor:regular-decomposition}

\end{corollary}

\begin{proof}
先求正则表示的特征标。固定基 $\{e_g\}$ 后，若 $h=e$，则 $L(h)=I$，故
\[
\chi_{\mathrm{reg}}(e)=|G|.
\]
若 $h\neq e$，则左乘 $g\mapsto hg$ 没有不动点，因此 $L(h)$ 对应的置换矩阵对角元全为零，从而
\[
\chi_{\mathrm{reg}}(h)=0.
\]
于是由重数公式，某不可约特征标 $\chi^{(\alpha)}$ 在正则表示中出现的重数为
\[
m_\alpha
=\frac1{|G|}\sum_{g\in G}\overline{\chi^{(\alpha)}(g)}\chi_{\mathrm{reg}}(g)
=\frac1{|G|}\overline{\chi^{(\alpha)}(e)}\cdot |G|
=d_\alpha.
\]
故正则表示分解如上。两边取维数即得
\[
|G|=\sum_\alpha d_\alpha^2.
\]
\end{proof}

实际使用重数公式时，通常先把逐群元求和改写成逐共轭类求和。若 $C_1,\dots,C_r$ 是共轭类，$g_s\in C_s$ 为代表元，则
\[
\langle\chi,\psi\rangle_G
=\frac1{|G|}\sum_{s=1}^r |C_s|\,\overline{\chi(g_s)}\psi(g_s).
\]
因此一张特征标表真正需要的列不是“每个群元”，而是“每个共轭类”。这也解释了\chapref{chap:homomorphism-quotient} 中共轭类的作用：它不是孤立的群结构知识，而是表示论计算的天然坐标。

把前面所有抽象结论汇总到一个具体群上，最合适的贯穿例子依然是 $S_3$。它有三个共轭类：
\[
C_1=\{e\},\qquad C_2=\{(12),(13),(23)\},\qquad C_3=\{(123),(132)\}.
\]
由\chapref{chap:group-basics}可知这三类的大小分别为 $1,3,2$。因此 $S_3$ 共有三个不等价不可约表示。由维数平方和公式
\[
6=d_1^2+d_2^2+d_3^2
\]
唯一可能是
\[
(d_1,d_2,d_3)=(1,1,2).
\]
两个一维表示分别是平凡表示与符号表示 $\operatorname{sgn}$；二维表示就是前面提到的标准表示。于是得到特征标表如下。

同一个群换成 $D_3$ 的几何记号以后，特征标表读起来更接近物理常用的 $C_{3v}$ 形式。$D_3$ 的三个共轭类为
\[
\{e\},\qquad \{r,r^2\},\qquad \{s,sr,sr^2\},
\]

\begin{pmatrix}

\end{pmatrix},
\qquad

\begin{pmatrix}
1&0\\0&-1
\end{pmatrix},
\label{eq:d3-E-matrices}
\end{equation}

$D^{(E)}(r)^3=D^{(E)}(s)^2=I$ 与

用类大小做加权内积可以逐行验证正交关系。例如
\[
\langle\chi_E,\chi_E\rangle_{D_3}
=\frac{1}{6}\bigl(1\cdot2^2+2\cdot(-1)^2+3\cdot0^2\bigr)=\frac{6}{6}=1,
\]
确认 $E$ 不可约；又如
\[
\langle\chi_{A_2},\chi_E\rangle_{D_3}
=\frac{1}{6}\bigl(1\cdot1\cdot2+2\cdot1\cdot(-1)+3\cdot(-1)\cdot0\bigr)=0,
\]
确认 $A_2$ 与 $E$ 不含公共不可约成分。其余行对同理。这张表与 $S_3$ 的特征标表\cref{tab:s3-character-table}只差一个同构映射 $r\leftrightarrow(123)$、$s\leftrightarrow(12)$，但几何记号让"旋转类"与"反射类"一目了然，这正是物理中 $C_{3v}$ 特征标表的标准写法。
特征标表不只是代数对象，它直接约束分子的振动简并结构。以氨分子 $\mathrm{NH_3}$ 为例：其 $C_3$ 主轴穿过氮原子和三个氢原子所成三角形的中心，另有三个 $\sigma_v$ 镜面，对称群为 $C_{3v}\cong D_3$。4 个原子各有 3 个位移分量，故\emph{位移表示} $\Gamma_{\mathrm{disp}}$ 是 $3N=12$ 维。它的特征标只需数"在群元作用下不动的原子"，因为只有不动原子才贡献对角元：每个不动原子贡献一个 $3\times3$ 正交矩阵的迹。
\begin{itemize}[leftmargin=3em]
\item $e$：4 个原子都不动，每个贡献 $\Tr I_3=3$，故 $\chi_{\mathrm{disp}}(e)=12$。
\item $r$（$C_3$）：3 个 H 原子循环置换、N 不动。N 贡献 $\Tr R_{\uv z}(120^\circ)=1+2\cos120^\circ=0$；被置换的 H 原子不贡献对角元。故 $\chi_{\mathrm{disp}}(r)=0$。
\item $s$（$\sigma_v$）：N 不动、1 个 H 在镜面上不动、2 个 H 互换。每个不动原子贡献 $\Tr\operatorname{diag}(1,1,-1)=1$。故 $\chi_{\mathrm{disp}}(s)=1+1=2$。
\end{itemize}
于是 $\chi_{\mathrm{disp}}=(12,0,2)$。从中扣除 3 个整体平移和 3 个整体转动，才得到真正的\emph{振动表示} $\Gamma_{\mathrm{vib}}$。平移表示即极矢量表示，转动表示即轴矢量表示，由特征标表读出
\[
\Gamma_{\mathrm{trans}}=A_1\oplus E,\quad \chi_{\mathrm{trans}}=(3,0,1),
\qquad
\Gamma_{\mathrm{rot}}=A_2\oplus E,\quad \chi_{\mathrm{rot}}=(3,0,-1).
\]
因此
\[
\chi_{\mathrm{vib}}=\chi_{\mathrm{disp}}-\chi_{\mathrm{trans}}-\chi_{\mathrm{rot}}
=(12,0,2)-(3,0,1)-(3,0,-1)=(6,0,2).
\]
用重数公式逐个提取不可约成分：
\begin{align*}
m_{A_1}&=\tfrac16\bigl(1\cdot1\cdot6+2\cdot1\cdot0+3\cdot1\cdot2\bigr)=\tfrac{12}{6}=2,\\
m_{A_2}&=\tfrac16\bigl(1\cdot1\cdot6+2\cdot1\cdot0+3\cdot(-1)\cdot2\bigr)=\tfrac{0}{6}=0,\\
m_E&=\tfrac16\bigl(1\cdot2\cdot6+2\cdot(-1)\cdot0+3\cdot0\cdot2\bigr)=\tfrac{12}{6}=2.
\end{align*}
于是
\begin{equation}
\Gamma_{\mathrm{vib}}=2A_1\oplus 2E.
\label{eq:nh3-vib-decomposition}
\end{equation}
$\mathrm{NH_3}$ 的 6 个振动自由度包含两份 $A_1$ 和两份 $E$。
在保持 $C_{3v}$ 对称性的谐振近似中，动力学矩阵可在每种类型的重数空间中继续对角化：
两份 $A_1$ 给出两个通常不同的频率，两份 $E$ 给出两组各至少二重简并的频率。
通常将它们分别命名为对称伸缩 $\nu_1$、对称弯曲 $\nu_2$、
简并伸缩 $\nu_{3a},\nu_{3b}$ 和简并弯曲 $\nu_{4a},\nu_{4b}$；
伸缩与弯曲的具体混合比例仍由力常数决定。
本模型不含 $A_2$ 振动，是因为上面的特征标内积为零，
不能由此推出一般分子的振动位移都排除 $A_2$ 类型。
群论在求解动力学以前确定这些类型及其重数；保持对称性的力常数变化
保留 $E$ 型的二重简并，但仍可能使不同类型或不同副本的频率偶然重合。
特征标表不只给出不可约分解，它本身还有丰富的结构。下面几条性质在物理应用中反复使用。
\begin{theorem}[特征标的列正交关系]\label{thm:character-column-orthogonality}
不可约特征标不仅按行正交，也按列正交：对任意 $g,h\in G$，
\[
\sum_{\alpha=1}^{k}\chi^{(\alpha)}(g)\,\overline{\chi^{(\alpha)}(h)}
=\begin{cases}
\lvert C_G(g)\rvert, & g\sim h,\\
0, & g\not\sim h.
\end{cases}
\]
\end{theorem}
\begin{proof}
大正交关系给出
\[
\frac1{\lvert G\rvert}\sum_{g\in G}\chi^{(\alpha)}(g)\overline{\chi^{(\beta)}(g)}=\delta_{\alpha\beta}.
\]
这是行正交：把特征标看作 $L^2(G)$ 中的函数，不同不可约表示的特征标正交。
把特征标表写成矩阵 $X$，行索引为不可约表示 $\alpha=1,\dots,k$，列索引为共轭类 $C_j$（$j=1,\dots,k$），则行正交写成
\[
\frac1{\lvert G\rvert}\sum_{g}\chi^{(\alpha)}(g)\overline{\chi^{(\beta)}(g)}
=\frac1{\lvert G\rvert}\sum_j\lvert C_j\rvert\,\chi^{(\alpha)}(C_j)\overline{\chi^{(\beta)}(C_j)}
=\delta_{\alpha\beta}.
\]
定义\emph{归一化特征标表} $\tilde X$，其矩阵元为
\[
\tilde X_{\alpha j}=\chi^{(\alpha)}(C_j)\sqrt{\frac{\lvert C_j\rvert}{\lvert G\rvert}}.
\]
则行正交关系变为
\[
\sum_j\tilde X_{\alpha j}\overline{\tilde X_{\beta j}}=\delta_{\alpha\beta},
\]
即 $\tilde X\,\tilde X^\dagger=I$，$\tilde X$ 是酉矩阵。
由于 $\tilde X$ 是方阵，$\tilde X\tilde X^\dagger=I$ 也蕴含 $\tilde X^\dagger\tilde X=I$，从而给出列正交：
\[
\sum_\alpha\overline{\tilde X_{\alpha i}}\,\tilde X_{\alpha j}=\delta_{ij}.
\]
展开归一化因子：
\[
\sum_\alpha\overline{\chi^{(\alpha)}(C_i)}\,\chi^{(\alpha)}(C_j)\,\frac{\sqrt{\lvert C_i\rvert\lvert C_j\rvert}}{\lvert G\rvert}=\delta_{ij}.
\]
对 $i=j$：
\[
\sum_\alpha\lvert\chi^{(\alpha)}(C_i)\rvert^2=\frac{\lvert G\rvert}{\lvert C_i\rvert}=\lvert C_G(g_i)\rvert,
\]
其中最后一步用了 $\lvert C_i\rvert=\lvert G\rvert/\lvert C_G(g_i)\rvert$（共轭类大小等于群阶除以中心化子阶）。
对 $i\neq j$：
\[
\sum_\alpha\overline{\chi^{(\alpha)}(C_i)}\,\chi^{(\alpha)}(C_j)=0.
\]
合并两种情形，对任意 $g,h\in G$ 有
\[
\sum_{\alpha=1}^{k}\chi^{(\alpha)}(g)\overline{\chi^{(\alpha)}(h)}
=\begin{cases}\lvert C_G(g)\rvert,&g\sim h,\\0,&g\not\sim h.\end{cases}
\]
这里 $g\sim h$ 表示共轭。当 $g=h$ 时给出 $\sum_\alpha\lvert\chi^{(\alpha)}(g)\rvert^2=\lvert C_G(g)\rvert$；当 $g,h$ 不共轭时特征标向量正交。
\end{proof}
因此不同共轭类的特征标向量彼此正交；这既能检验特征标表是否遗漏不可约表示，也使
$\sum_\alpha\lvert\chi^{(\alpha)}(g)\rvert^2=\lvert C_G(g)\rvert$
成为从表中读取中心化子阶的直接判据。
\begin{proposition}[特征标的实性判据]
设 $\chi$ 是不可约特征标。定义
\[
\nu(\chi)=\frac1{\lvert G\rvert}\sum_{g\in G}\chi(g^2).
\]
则：
\begin{enumerate}[label=(\roman*),leftmargin=3em]
\item $\nu(\chi)=1$ 当且仅当 $\chi$ 取实值且表示可写成实矩阵；
\item $\nu(\chi)=-1$ 当且仅当 $\chi$ 取实值但表示不可写成实矩阵（ 四元数型）；
\item $\nu(\chi)=0$ 当且仅当 $\chi$ 不取实值（复型）。
\end{enumerate}
\label{prop:character-reality}
\end{proposition}
该判据的完整分类及独立证明见\cref{thm:frobenius-schur-classification}；
那里从群平均的不变张量出发，构造平方为 $+I$ 或 $-I$ 的反酉映射，
因而区分实矩阵实现与四元数结构。
这条判据在物理中用于判断表示能否用实矩阵实现——这决定了相应的简并态能否用实波函数描述。$D_3$ 的 $A_1,A_2$ 是实一维表示，二维表示 $E$ 也已由实矩阵实现；三者的指标均为 $\nu=1$，故都可实化。
\begin{proposition}[正则表示的特征标与投影]
正则表示的特征标为
\[
\chi_{\mathrm{reg}}(e)=\lvert G\rvert,
\qquad
\chi_{\mathrm{reg}}(g)=0\quad(g\neq e).
\]
利用正交关系，正则表示分解为
\[
\rho_{\mathrm{reg}}\simeq\bigoplus_{\alpha=1}^{k}d_\alpha\,\rho^{(\alpha)},
\qquad
d_\alpha=\dim\rho^{(\alpha)}.
\]
\label{prop:regular-character}
\end{proposition}
正则表示就是群代数 $\CC[G]$ 在自身上的左乘作用。上述分解说明群代数作为表示空间被拆成 $k$ 个不可约块，第 $\alpha$ 块出现 $d_\alpha$ 次。这正是 Wedderburn 分解
\[
\CC[G]\cong\bigoplus_{\alpha=1}^{k}M_{d_\alpha}(\CC)
\]
的表示论翻译：群代数同构于若干个全矩阵代数的直和。这一结构在后续构造投影算符和对称性适配线性组合时直接使用。
限制和诱导是表示论中两个互相对偶的操作：限制把大群的表示缩小到子群，诱导把子群的表示扩大到大群。Frobenius 互反律说明这两个操作在内积下"伴随"。
\begin{definition}[限制表示与诱导表示]
设 $G$ 为有限群、$H\leqslant G$，$\rho:G\to\operatorname{GL}(V)$ 为 $G$ 的表示。
把 $\rho$ 限制到 $H$ 上，得到 $H$ 的表示 $\rho\downarrow_H^G$；空间仍为 $V$，作用为 $\rho(h)$（$h\in H$），特征标为 $\chi\downarrow_H^G(h)=\chi(h)$。
反过来，设 $\sigma:H\to\operatorname{GL}(W)$ 为 $H$ 的表示，并取左陪集代表 $g_1,\dots,g_r$（$r=[G:H]$）。诱导表示空间为
\[
\operatorname{Ind}_H^G W=\bigoplus_{i=1}^{r}g_i\otimes W,
\]
把 $g_i\otimes w$ 视为第 $i$ 个 $W$ 副本中的向量。若 $gg_i=g_jh$（$h\in H$），定义
\[
g\cdot(g_i\otimes w)=g_j\otimes\sigma(h)w.
\]
陪集分解使 $j,h$ 唯一。维数满足 $\dim\operatorname{Ind}_H^G W=[G:H]\cdot\dim W$。
\label{def:restriction-induction}
\end{definition}
\begin{proposition}[诱导作用的相容性]
上述公式定义一个 $G$ 的表示；不同陪集代表的选择给出等价表示。
\end{proposition}

\begin{proof}
若 $bg_i=g_jh_1$、$ag_j=g_kh_2$，则
$(ab)g_i=g_kh_2h_1$。先作用 $b$ 再作用 $a$，向量在第 $k$ 块变成
$\sigma(h_2)\sigma(h_1)w=\sigma(h_2h_1)w$，恰与直接作用 $ab$ 相同。
单位元恒等作用，$g^{-1}$ 给出逆作用，所以这些线性变换可逆。
若改选代表 $g_i'=g_i h_i$，映射
$g_i'\otimes w\mapsto g_i\otimes\sigma(h_i)w$ 是可逆的，
并将两个作用公式联系起来；重排陪集仅重排各直和块。
\end{proof}

\begin{theorem}[Frobenius 互反律]
设 $G$ 为有限群、$H\leqslant G$，$\sigma$ 为 $H$ 的表示，$\rho$ 为 $G$ 的表示。则
\[
\left\langle\chi_{\operatorname{Ind}_H^G\sigma},\,\chi_\rho\right\rangle_G
=
\left\langle\chi_\sigma,\,\chi_{\rho\downarrow_H^G}\right\rangle_H.
\]
当 $\sigma,\rho$ 不可约时，这表示 $\sigma$ 在 $\rho\downarrow_H^G$ 中的重数，等于 $\rho$ 在 $\operatorname{Ind}_H^G\sigma$ 中的重数。对于一般表示，等式给出交织映射空间维数的相等。
\label{thm:frobenius-reciprocity}
\end{theorem}
\par\medskip
\begin{proof}[证明（Frobenius 互反律）]
诱导特征标可直接由陪集模型求迹。取左陪集代表 \(G=\bigsqcup_i g_iH\)。元素 \(g\) 只在满足 \(g_i^{-1}gg_i\in H\) 的陪集块上留下对角贡献，因此
\begin{equation}
\chi_{\operatorname{Ind}_H^G\sigma}(g)
=\sum_{\substack{i\\g_i^{-1}gg_i\in H}}
\chi_\sigma(g_i^{-1}gg_i)
=\frac1{|H|}
\sum_{\substack{x\in G\\x^{-1}gx\in H}}
\chi_\sigma(x^{-1}gx).
\label{eq:induced-char-coset}
\end{equation}
第二个等号来自每个陪集含 \(|H|\) 个元素，而特征标在 \(H\) 的共轭类上不变。将它代入 \(G\) 上的特征标内积，并令 \(h=x^{-1}gx\)，得到
\begin{align*}
\left\langle\chi_{\operatorname{Ind}\sigma},\chi_\rho\right\rangle_G
&=\frac1{|G||H|}
\sum_{x\in G}\sum_{h\in H}
\overline{\chi_\sigma(h)}\chi_\rho(xhx^{-1})\\
&=\frac1{|H|}\sum_{h\in H}
\overline{\chi_\sigma(h)}\chi_\rho(h)
=\left\langle\chi_\sigma,\chi_{\rho\downarrow_H^G}\right\rangle_H .
\end{align*}
这里只用了变量替换和 \(\chi_\rho(xhx^{-1})=\chi_\rho(h)\)，故重数等式随即成立。
对于一般表示，交织空间之间还存在下面的显式同构。给定
$T\in\operatorname{Hom}_G(\operatorname{Ind}_H^G W,V)$，取
$F(w)=T(e\otimes w)$，便得到 $H$-交织映射。
反过来，给定 $F\in\operatorname{Hom}_H(W,V\downarrow_H)$，定义
\[
T(g_i\otimes w)=\rho(g_i)F(w).
\]
若 $gg_i=g_jh$，则
$T(g\cdot(g_i\otimes w))=\rho(g_j)F(\sigma(h)w)
=\rho(g_jh)F(w)=\rho(g)T(g_i\otimes w)$，所以 $T$ 等变。
两种构造互逆，证明自然同构
\[
\operatorname{Hom}_G(\operatorname{Ind}_H^G W,V)
\cong\operatorname{Hom}_H(W,V\downarrow_H).
\]
\end{proof}

当 $W,V$ 都不可约时，右侧告诉我们 $W$ 在限制表示中出现的次数；
左侧则在诱导表示中读出 $V$ 的重数。两种计算实际上使用同一组交织映射。

\begin{example}[从 $C_3$ 的相位诱导出 $D_3$ 的二维表示]
取 $H=C_3=\langle r\rangle\triangleleft D_3$，令
$\sigma_1(r)=\omega=\ee^{2\pi\ii/3}$。求诱导表示的矩阵及不可约分解。

\textbf{解。} 选陪集代表 $e,s$，基记为 $e\otimes1,s\otimes1$。
由 $rs=sr^{-1}$，$r$ 在这两个块分别作用为 $\omega$ 与 $\omega^{-1}$；$s$ 交换两块。
因此
\[
D(r)=\begin{pmatrix}\omega&0\\0&\omega^{-1}\end{pmatrix},\qquad
D(s)=\begin{pmatrix}0&1\\1&0\end{pmatrix}.
\]
特征标在 $e,r,s$ 上为 $2,-1,0$，且
$\langle\chi,\chi\rangle=(4+2)/6=1$，所以诱导表示不可约，等价于 $E$。
限制回 $C_3$ 时两条坐标轴分别承载 $\sigma_1,\sigma_1^*$，故
$E\downarrow_{C_3}=\sigma_1\oplus\sigma_1^*$；互反律两边的相应重数都是 $1$。
\end{example}

\begin{theorem}[Mackey 限制分解]
设 $H,K\le G$，$G$ 有限，$\sigma$ 是 $H$ 的表示。取双陪集 $KdH$ 的代表 $d$，则
\begin{equation}
\left(\operatorname{Ind}_H^G\sigma\right)\downarrow_K
\cong\bigoplus_{d\in K\backslash G/H}
\operatorname{Ind}_{K\cap dHd^{-1}}^K
\left({}^{d}\sigma\downarrow_{K\cap dHd^{-1}}\right),
\label{eq:mackey-restriction}
\end{equation}
其中 ${}^{d}\sigma(x)=\sigma(d^{-1}xd)$，定义域为 $dHd^{-1}$。
\end{theorem}

\begin{proof}
$K$ 在陪集集合 $G/H$ 上的轨道恰为双陪集 $KdH$ 中的左陪集。
诱导空间中由每个轨道的陪集块张成的空间分别 $K$-不变，故按双陪集直和分解。
固定 $d$，点 $dH$ 的稳定子为 $L=K\cap dHd^{-1}$；
若 $l\in L$，则 $ld=d(d^{-1}ld)$，所以 $l$ 在 $d\otimes W$ 上作用为 $\sigma(d^{-1}ld)$。
将此块向整个 $K$-轨道搬运，映射
\[
k\otimes w\longmapsto kd\otimes w
\]
便给出该轨道空间与 $\operatorname{Ind}_L^K({}^{d}\sigma|_L)$ 的同构。
关系 $kl\otimes w=k\otimes{}^{d}\sigma(l)w$ 在右侧也成立，保证良定义；
轨道--稳定子双射使陪集块一一对应，故映射双射并保持 $K$ 的作用。
各轨道相加即得分解。
\end{proof}

\section{紧群上的平均与完备展开}
有限群的平均用每个群元相同的权重。若群元连续变化，就需要一种不因整体左移或右移而改变的积分。这里的紧群指紧 Hausdorff 拓扑群：乘法与求逆连续，任意开覆盖有有限子覆盖，且不同点可用不交开集分离。紧矩阵群采用矩阵元素的通常拓扑，连续表示则要求每个矩阵元都是连续函数。

\begin{definition}[归一化 Haar 测度与函数内积]
紧群 $G$ 的归一化 Haar 测度是一个正规 Borel 概率测度 $\mu$，满足 $\mu(hA)=\mu(Ah)=\mu(A)$。正规性指可由紧集从内、开集从外逼近可测集的测度。记 $\dd g=\dd\mu(g)$，则 $\int_G\dd g=1$。平方可积函数按几乎处处相等识别，组成 Hilbert 空间 $L^2(G)$；本节取
\[
\langle f,h\rangle_G=\int_G\overline{f(g)}h(g)\,\dd g.
\]
因此内积对第二变量线性。有限群给出 $\int_G f\,\dd g=|G|^{-1}\sum_gf(g)$；圆群 $U(1)=\{\ee^{\ii\phi}\}$ 给出 $\dd g=\dd\phi/(2\pi)$。
\label{def:compact-haar-normalization}
\end{definition}

本节采用 Haar 存在唯一性定理作为测度论输入：每个紧 Hausdorff 群存在唯一归一化左不变正规 Borel 测度，它也右不变并在求逆下不变。后两性质可由唯一性检查：右移的推前仍是归一化左不变测度；求逆交换左、右不变性。本节同时采用 Stone--Weierstrass 逼近定理：紧 Hausdorff 空间上的连续复函数子代数，若含常数、对共轭封闭且能分离不同点，则在一致范数下稠密。紧 Lie 群的 Haar 构造及这一逼近定理的表述见\href{https://courses.washington.edu/mnotes/liegroups/notes.pdf}{华盛顿大学紧 Lie 群讲义，第 37--38、61 页}；一般紧群的 Haar 存在性属于测度论，此处不从有限群计数推出。

有了不变积分，有限群的平均证明仍能逐行使用。若 $\rho$ 是有限维连续表示，用 $\int_G\langle\rho(g)v,\rho(g)w\rangle\,\dd g$ 酉化；被积函数对非零 $v$ 严格为正，积分有限且正。群不变子空间的正交补仍群不变，所以表示完全可约。

\begin{proposition}[紧群矩阵元的正交关系]
对两组不等价有限维连续不可约酉表示 $\pi_\alpha,\pi_\beta$，在正交归一基下有
\[
\int_G\overline{\pi_\alpha^{ij}(g)}\pi_\beta^{kl}(g)\,\dd g
=\frac{\delta_{\alpha\beta}\delta_{ik}\delta_{jl}}{d_\alpha}.
\label{eq:compact-matrix-orthogonality}
\]
\end{proposition}

\begin{proof}
对任意 $A:V_\alpha\to V_\beta$，令 $T(A)=\int_G\pi_\beta(g)A\pi_\alpha(g)^{-1}\,\dd g$。以 $g\mapsto hg$ 换元可见 $T(A)$ 交织两表示；Schur 引理给出不同类型时 $T(A)=0$，同一类型时 $T(A)=cI$。后一情形取迹得到 $c=\operatorname{Tr}A/d_\alpha$。取仅第 $(l,j)$ 个元素为 $1$ 的 $A$，再读取 $T(A)$ 的 $(k,i)$ 元；酉性使 $[\pi_\alpha(g)^{-1}]_{ji}=\overline{\pi_\alpha^{ij}(g)}$，即得公式。对角矩阵元求和又给出不可约特征标的正交性。
\end{proof}

\begin{theorem}[Peter--Weyl 完备性]
对紧群 $G$，全部不等价有限维连续不可约酉表示的函数 $\sqrt{d_\alpha}\,\pi_\alpha^{ij}$ 构成 $L^2(G)$ 的正交完备系。有限矩阵元线性组合还在连续函数空间 $C(G)$ 中一致稠密。
\label{thm:compact-peter-weyl}
\end{theorem}

\begin{proof}[证明（紧矩阵群情形）]
设 $G$ 是某个 $\operatorname{GL}(n,\CC)$ 的紧子群。令 $\mathcal A$ 为所有有限维连续表示矩阵元的线性张成。平凡表示使它含常数；矩阵元的乘积是张量积表示的矩阵元，复共轭是共轭表示的矩阵元，所以它是对共轭封闭的子代数。定义表示 $g\mapsto g$ 已能分离不同矩阵，因而 $\mathcal A$ 分离点。Stone--Weierstrass 定理给出 $\overline{\mathcal A}^{\|\cdot\|_\infty}=C(G)$。

每个有限维表示完全可约，因此 $\mathcal A$ 也由不可约表示矩阵元张成。正规有限测度下，连续函数在 $L^2$ 中稠密；且 $\|f\|_2\le\|f\|_\infty$，所以同一矩阵元张成空间在 $L^2(G)$ 中稠密。结合 \eqnrefs{eq:compact-matrix-orthogonality} 得到完备性。这里的一致稠密是说存在某些有限线性组合逼近连续函数，并未声称任意排列的 Fourier 部分和一致收敛。
\end{proof}

对不预先给定忠实有限维矩阵实现的一般紧群，矩阵元能否分离点也须证明；这部分采用一般 Peter--Weyl 定理作为输入。紧 Lie 群中从紧自伴卷积算子的谱分解建立分离性的证明见\href{https://ocw.mit.edu/courses/18-755-lie-groups-and-lie-algebras-ii-spring-2024/mit18_755_s24_lec10.pdf}{MIT 18.755 第 10 讲，第 36 节}。上述独立证明已覆盖本书实际使用的紧矩阵群；$SO(3)$ 的具体矩阵元将在角动量分类以后代入 \cref{thm:compact-peter-weyl}。
记 $\pi_\alpha^{ij}(g)$ 为不可约表示 $\alpha$ 在一组正交归一基下的 $(i,j)$ 矩阵元，Fourier 系数定义为 $\widehat f_\alpha^{ij}=\int_G\overline{\pi_\alpha^{ij}(g)}\,f(g)\,\dd g$，矩阵 $\widehat f_\alpha=(\widehat f_\alpha^{ij})$ 的 Hilbert--Schmidt 范数平方为 $\|\widehat f_\alpha\|_{\mathrm{HS}}^2=\sum_{i,j}|\widehat f_\alpha^{ij}|^2$。
\begin{proposition}[Plancherel 公式]
设 $G$ 为紧群，Haar 测度归一化为 $\int_G\dd g=1$，$f\in L^2(G)$。则
\[
\|f\|_{L^2}^2=\int_G|f(g)|^2\,\dd g
=\sum_\alpha d_\alpha\|\hat f_\alpha\|_{\text{HS}}^2
=\sum_\alpha d_\alpha\sum_{i,j}|\hat f_\alpha^{ij}|^2.
\]
\end{proposition}

\begin{proof}
Peter--Weyl 展开在 \(L^2\) 意义下为
\[
f(g)=\sum_\alpha d_\alpha\sum_{i,j}
\widehat f_\alpha^{ij}\pi_\alpha^{ij}(g).
\]
先对有限部分和计算范数，再取 \(L^2\) 极限。矩阵元正交关系消去所有 \(\alpha\neq\beta\) 或指标不同的交叉项：
\begin{align*}
\|f\|_2^2
&=\sum_{\alpha,\beta}d_\alpha d_\beta
\sum_{i,j,k,l}
\overline{\widehat f_\alpha^{ij}}\widehat f_\beta^{kl}
\int_G\overline{\pi_\alpha^{ij}(g)}\pi_\beta^{kl}(g)\,\dd g\\
&=\sum_\alpha d_\alpha\sum_{i,j}
|\widehat f_\alpha^{ij}|^2
=\sum_\alpha d_\alpha\|\widehat f_\alpha\|_{\mathrm{HS}}^2.
\end{align*}
有限部分和在 \(L^2\) 中收敛，故等式延拓到任意 \(f\in L^2(G)\)。
\end{proof}

\begin{example}[圆群上的展开与范数]
在 $U(1)$ 上取归一化测度 $\dd\phi/(2\pi)$。求 $f(\phi)=1+2\cos\phi+\sin2\phi$ 的全部不可约表示分量，并核对 Plancherel 范数。

\textbf{解。} 连续一维不可约表示是 $\pi_n(\ee^{\ii\phi})=\ee^{\ii n\phi}$（$n\in\ZZ$）。将三角函数写成指数，得到
\[
f(\phi)=1+\ee^{\ii\phi}+\ee^{-\ii\phi}
-\frac{\ii}{2}\ee^{2\ii\phi}+\frac{\ii}{2}\ee^{-2\ii\phi}.
\]
因此 $\widehat f_0=\widehat f_{1}=\widehat f_{-1}=1$、$\widehat f_2=-\ii/2$、$\widehat f_{-2}=\ii/2$，其余为零。每个表示维数为一，故范数为 $1+1+1+1/4+1/4=7/2$。直接积分时不同频率的交叉项消失，$\cos^2\phi$ 与 $\sin^22\phi$ 的平均均为 $1/2$，也得到 $1+4/2+1/2=7/2$。不可约表示的矩阵元在这里就是熟悉的 Fourier 模式；后面的旋转群应用会把单个整数 $n$ 换成矩阵的两个磁量子数指标。
\end{example}
