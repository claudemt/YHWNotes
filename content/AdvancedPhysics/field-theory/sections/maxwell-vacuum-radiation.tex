\subsection{真空响应与辐射通道}
\label{sec:feqo-vacuum-dyadic}
\begin{proposition}[真空并矢与纵横分解]
对 $\omega>0$ 取 $k=(\omega+\ii0^+)/c$，$R=|\mathbf r-\mathbf r'|$。真空推迟核是
\begin{equation}
 \mathbf G_0^R(\mathbf r,\mathbf r';\omega)
 =\left(\mathbf I+\frac{\nabla\nabla}{k^2}\right)
 \frac{\ee^{\ii kR}}{4\pi R}.
 \label{eq:feqo-vacuum-green}
\end{equation}
其中二阶导数按分布定义。若空间 Fourier 变量为 $\mathbf q$，定义 $P_L=\mathbf q\mathbf q^{\mathsf T}/q^2$、$P_T=\mathbf I-P_L$，则
\begin{equation}
 \mathbf G_0^R(\mathbf q,\omega)
 =\frac{P_T}{q^2-k^2}-\frac{P_L}{k^2}.
 \label{eq:feqo-vacuum-transverse-longitudinal}
\end{equation}
完整 Green 张量包含纵向静电响应；只有源点外的自由传播辐射满足横向条件。
\end{proposition}

\begin{proof}
标量核 $g_0=\ee^{\ii kR}/(4\pi R)$ 满足 $(\nabla^2+k^2)g_0=-\delta^{(3)}(\mathbf r-\mathbf r')$，其 $1/(4\pi)$ 系数由包围源点的小球通量固定。令 $\mathsf L_0=\nabla\times\nabla\times-k^2\mathbf I$，逐项作用于并矢：
\begin{align*}
 \mathsf L_0(\mathbf I g_0)&=\nabla\nabla g_0-\mathbf I(\nabla^2+k^2)g_0,\\
 \mathsf L_0(k^{-2}\nabla\nabla g_0)&=-\nabla\nabla g_0.
\end{align*}
相加得到 $\mathsf L_0\mathbf G_0^R=\mathbf I\delta^{(3)}$。在空间 Fourier 表象，$\mathsf L_0=(q^2-k^2)P_T-k^2P_L$，两个正交子空间分别求逆便得到\eqnrefs{eq:feqo-vacuum-transverse-longitudinal}。取散度也显示接触项的作用：
\begin{equation*}
 \nabla\cdot\mathbf G_0^R
 =\nabla(g_0+k^{-2}\nabla^2g_0)
 =-k^{-2}\nabla\delta^{(3)}(\mathbf r-\mathbf r').
\end{equation*}
因此“完整并矢在分布意义下处处无散度”并不成立。
\end{proof}

\begin{exbox}[点电偶极与真空辐射功率]
采用单频实场约定 $\mathbf E_{\rm real}=\operatorname{Re}[\boldsymbol{\mathcal E}\ee^{-\ii\omega t}]$。位于 $\mathbf r_0$ 的电偶极幅度 $\mathbf d$（单位 $\mathrm{C\,m}$）产生
\begin{equation*}
 \mathbf J=-\ii\omega\mathbf d\,\delta^{(3)}(\mathbf r-\mathbf r_0),
 \qquad \rho=-\mathbf d\cdot\nabla\delta^{(3)}(\mathbf r-\mathbf r_0),
 \qquad -\ii\omega\rho+\nabla\cdot\mathbf J=0.
\end{equation*}
将这个守恒源代入\eqnrefs{eq:E-from-GJ-SI-r12}，得 $\boldsymbol{\mathcal E}=\mu_0\omega^2\mathbf G_0^R\mathbf d$。在 $kR\gg1$ 的远场，
\begin{equation*}
 \mathbf G_0^R=\frac{\ee^{\ii kR}}{4\pi R}
 (\mathbf I-\hat{\mathbf R}\hat{\mathbf R}^{\mathsf T})+O(R^{-2}).
\end{equation*}
故 $\mathbf d_\perp=\mathbf d-\hat{\mathbf R}(\hat{\mathbf R}\cdot\mathbf d)$ 决定辐射方向，时间平均 Poynting 通量给
\begin{equation}
 \frac{\dd P_{\rm rad}}{\dd\Omega}
 =\frac{\omega^4}{32\pi^2\varepsilon_0c^3}|\mathbf d_\perp|^2,
 \qquad
 P_{\rm rad}=\frac{\omega^4|\mathbf d|^2}{12\pi\varepsilon_0c^3}.
 \label{eq:feqo-vacuum-dipole-radiation}
\end{equation}
最后一步用了 $\int\dd\Omega\,\hat R_i\hat R_j=4\pi\delta_{ij}/3$。实场幅度中的 $1/2$ 时间平均因子必须保留；若把正频分量的幅度直接当成 $\boldsymbol{\mathcal E}$，功率会错四倍。

同一个功率还能从局域谱权重获得。真空动量核的横向部分满足
\begin{equation*}
 \operatorname{Im}\mathbf G_0^R(\mathbf r_0,\mathbf r_0;\omega)
 =\pi\int\frac{\dd^3q}{(2\pi)^3}P_T(\mathbf q)\delta(q^2-k^2)
 =\frac{k}{6\pi}\mathbf I.
\end{equation*}
这里径向积分为 $k/2$，横向投影的角积分为 $8\pi\mathbf I/3$。于是
\begin{equation*}
 \frac{\mu_0\omega^3}{2}\mathbf d^*\cdot
 \operatorname{Im}\mathbf G_0^R(\mathbf r_0,\mathbf r_0;\omega)\mathbf d
 =\frac{\omega^4|\mathbf d|^2}{12\pi\varepsilon_0c^3}.
\end{equation*}
材料完全无损时这个谱权重仍非零：它记录光子流向无穷远的辐射通道。
\end{exbox}

\begin{proposition}[源功率、吸收与辐射边界]
对频率 $\omega>0$ 的紧支撑电流幅度 $\mathbf J$，记内积为 $\langle\mathbf u,\mathbf v\rangle=\int\dd^3r\,\mathbf u^*\cdot\mathbf v$，并定义算符虚部 $\operatorname{Im}_{H}G=(G-G^\dagger)/(2\ii)$。源交给电磁系统的时间平均功率是
\begin{equation}
 P_{\rm src}=-\frac12\operatorname{Re}\langle\mathbf J,\boldsymbol{\mathcal E}\rangle
 =\frac{\mu_0\omega}{2}\langle\mathbf J,\operatorname{Im}_{H}\mathbf G^R\mathbf J\rangle
 =P_{\rm abs}+P_{\rm rad}.
 \label{eq:feqo-maxwell-power-balance}
\end{equation}
其中 $P_{\rm abs}$ 是材料内的吸收，$P_{\rm rad}$ 是流向无穷远的功率。在被动材料及出射边界下，两者均非负；有限探测孔径只收集 $P_{\rm rad}$ 的一部分。
\end{proposition}

\begin{proof}
将 $\boldsymbol{\mathcal E}=\ii\mu_0\omega\mathbf G^R\mathbf J$ 代入第一项就得到 Green 二次型。对场占据的体积 $V$ 使用
\begin{equation*}
 \nabla\cdot(\boldsymbol{\mathcal E}\times\boldsymbol{\mathcal H}^*)
 =\boldsymbol{\mathcal H}^*\cdot\nabla\times\boldsymbol{\mathcal E}
 -\boldsymbol{\mathcal E}\cdot\nabla\times\boldsymbol{\mathcal H}^*,
\end{equation*}
代入频域 Maxwell 方程并取实部，得
\begin{align*}
 P_{\rm src}
 &=\frac12\operatorname{Re}\int_{\partial V}
 (\boldsymbol{\mathcal E}\times\boldsymbol{\mathcal H}^*)\cdot\mathbf n\,\dd A\\
 &\quad+\frac{\omega\varepsilon_0}{2}\int_V\dd^3r\,
 \boldsymbol{\mathcal E}^*\cdot\operatorname{Im}_{H}\boldsymbol\varepsilon_r\boldsymbol{\mathcal E}
 +\frac{\omega\mu_0}{2}\int_V\dd^3r\,
 \boldsymbol{\mathcal H}^*\cdot\operatorname{Im}_{H}\boldsymbol\mu_r\boldsymbol{\mathcal H}.
\end{align*}
被动性使两个体积分非负。令边界扩到所有材料外的远场，出射条件使表面积分非负。有限体积中的通量则可有任意符号，不能直接解释为吸收。
\end{proof}

\begin{definition}[开放 Maxwell 逆核的辐射谱项]
在无磁损耗且外部为真空时，定义 $\mathsf R_{\rm rad}(\omega)$ 为辐射边界通道对 Green 算符虚部的贡献。其二次型由\eqnrefs{eq:feqo-maxwell-power-balance} 唯一固定为
\begin{equation}
 \langle\mathbf J,\mathsf R_{\rm rad}\mathbf J\rangle
 =\frac{2P_{\rm rad}[\mathbf J]}{\mu_0\omega}\ge0.
 \label{eq:feqo-radiative-spectral-operator}
\end{equation}
因而完整光学恒等式为
\begin{equation}
 \operatorname{Im}_{H}\mathbf G^R
 =\frac{\omega^2}{c^2}(\mathbf G^R)^\dagger
 \operatorname{Im}_{H}\boldsymbol\varepsilon_r\mathbf G^R
 +\mathsf R_{\rm rad}.
 \label{eq:feqo-open-green-optical-identity}
\end{equation}
在封闭无辐射边界下，或先让整个外部空间带上趋于零的吸收并在积分后取极限时，辐射项才可合并到体吸收表达式中。
\end{definition}

\begin{proof}
材料吸收满足
\begin{equation*}
 P_{\rm abs}=\frac{\omega\varepsilon_0}{2}\|\boldsymbol\varepsilon_{r,I}^{1/2}
 \boldsymbol{\mathcal E}\|^2
 =\frac{\mu_0\omega}{2}\left\langle\mathbf J,
 \frac{\omega^2}{c^2}(\mathbf G^R)^\dagger
 \boldsymbol\varepsilon_{r,I}\mathbf G^R\mathbf J\right\rangle.
\end{equation*}
这里 $\boldsymbol\varepsilon_{r,I}=\operatorname{Im}_{H}\boldsymbol\varepsilon_r$，且用了 $\mu_0\varepsilon_0c^2=1$。由功率守恒对任意紧支撑源成立，再以极化恒等式恢复双线性矩阵元，便得到算符恒等式。真空中体吸收为零，\eqnrefs{eq:feqo-vacuum-dipole-radiation} 的非零辐射功率完全来自 $\mathsf R_{\rm rad}$。
\end{proof}
