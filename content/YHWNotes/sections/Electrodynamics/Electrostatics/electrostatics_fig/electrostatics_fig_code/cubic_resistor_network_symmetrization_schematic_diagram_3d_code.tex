\begin{tikzfig}[
  x={(0.95cm,-0.1cm)},
  y={(0.40cm,0.5cm)},
  z={(0.00cm,0.85cm)}
]

  \def\LegendDrawLattice#1#2#3#4#5{

    \foreach \x in {0,...,#1}{
      \foreach \y in {0,...,#2}{
        \foreach \z in {0,...,#3}{
          \ifnum\x<#1
            \draw[#5] (\x,\y,\z)--(\the\numexpr\x+1\relax,\y,\z);
          \fi
          \ifnum\y<#2
            \draw[#5] (\x,\y,\z)--(\x,\the\numexpr\y+1\relax,\z);
          \fi
          \ifnum\z<#3
            \draw[#5] (\x,\y,\z)--(\x,\y,\the\numexpr\z+1\relax);
          \fi
        }
      }
    }

    \foreach \x in {0,...,#1}{
      \foreach \y in {0,...,#2}{
        \foreach \z in {0,...,#3}{
          \fill[black] (\x,\y,\z) circle (#4);
        }
      }
    }
  }

  \begin{scope}
    \LegendDrawLattice{1}{1}{1}{1.2pt}{black!60}

    \draw[green!70!black, line width=0.7pt] (0,0,0) circle (0.22);
    \draw[red!80!black,   line width=0.7pt,dashed] (1,1,1) circle (0.22);

    \begin{scope}[x={(1cm,0cm)},y={(0cm,1cm)},z={(0cm,0cm)}]
      \node[below] at (0.8,-0.9) {original $2\times 2\times 2$};
    \end{scope}
  \end{scope}

  \begin{scope}[xshift=3.2cm]
    \LegendDrawLattice{3}{3}{3}{1.0pt}{black!60}

    \foreach \xx in {0,3}{
      \foreach \yy in {0,3}{
        \foreach \zz in {0,3}{
          \draw[green!70!black, line width=0.7pt] (\xx,\yy,\zz) circle (0.15);
        }
      }
    }

    \foreach \xx in {1,2}{
      \foreach \yy in {1,2}{
        \foreach \zz in {1,2}{
          \draw[red!80!black, line width=0.7pt,dashed] (\xx,\yy,\zz) circle (0.15);
        }
      }
    }

    \begin{scope}[x={(1cm,0cm)},y={(0cm,1cm)},z={(0cm,0cm)}]
      \node[below] at (1.6,-0.9) {symmetrized $4\times 4\times 4$};
    \end{scope}
  \end{scope}
