\[Gamma][t_] := 1/2 Log[q/(q - 2)]*(1 + t)^-1;
P[m_, t_] := (1 - m^2)/(
  2 (1 + Sqrt[m^2 + (1 - m^2) (q/(q - 2))^(2/(1 + t))]));
P11[m_, t_] := (1 + m)/2 - P[m, t]; P00[m_, t_] := (1 - m)/2 - P[m, t];
\[Phi][m_, t_] := -(q/4) Log[q/(q - 2)] (1 - 4 P[m, t]) - 
   h m + (1 + 
      t) (q/2 (P11[m, t] Log[P11[m, t]] + P00[m, t] Log[P00[m, t]] + 
         2 P[m, t] Log[P[m, t]]) - (q - 
         1) ((1 + m)/2 Log[(1 + m)/2] + (1 - m)/2 Log[(1 - m)/2]));
FullSimplify[Series[\[Phi][m, t], {m, 0, 4}, {t, 0, 2}], 
 Assumptions -> q > 3]
FullSimplify[
 Series[-(1/2) m^2 - 
   h  m + (1 + 
      t) ((1 + m)/2 Log[ (1 + m)/2] + (1 - m)/2 Log[ (1 - m)/2]), {m, 
   0, 5}, {t, 0, 2}]]
