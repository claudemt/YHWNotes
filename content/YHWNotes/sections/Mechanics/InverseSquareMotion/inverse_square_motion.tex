\chapter{一次反比势场中的运动}
\section{在一次反比吸引势场中的运动}
假设空间中由固定源产生一次反比势能分布
\begin{align*}
	V(r) = -\dfrac{\alpha}{r}, \quad \alpha > 0.
\end{align*}

设运动的作用量为 \( S \)，则哈密顿 - 雅可比方程给出：
\begin{align*}
	\left( \dfrac{\partial S}{\partial r} \right)^2 + \left( \dfrac{\partial S}{r \partial \phi} \right)^2 + m^2 c^2 = \dfrac{1}{c^2} \left( E + \dfrac{\alpha}{r} \right)^2.
\end{align*}
分离变量，设 \( S = -Et + L\phi + f(r) \)，得：
\begin{align*}
	\left( \dfrac{\dd f(r)}{\dd r} \right)^2 = \dfrac{1}{c^2} \left( E + \dfrac{\alpha}{r} \right)^2 - \dfrac{L^2}{r^2} - m^2 c^2.
\end{align*}
因此
\begin{align*}
	f(r) = \int \sqrt{\dfrac{1}{c^2} \left( E + \dfrac{\alpha}{r} \right)^2 - \dfrac{L^2}{r^2} - m^2 c^2} \, \dd r,\quad
	S = -Et + L\phi + \int \sqrt{\dfrac{1}{c^2} \left( E + \dfrac{\alpha}{r} \right)^2 - \dfrac{L^2}{r^2} - m^2 c^2} \, \dd r.
\end{align*}
代入正则方程，并适当选择角度和时间零点，使得常数为 0：
\begin{align*}
	\dfrac{\partial S}{\partial L} = \phi_0 = 0, \quad \dfrac{\partial S}{\partial E} = t_0 = 0.
\end{align*}
得到 \( \phi(r) \) 和 \( t(r) \) 的函数关系：
\begin{align*}
	\phi &= \int \dfrac{cL}{\sqrt{(\alpha^2 - c^2 L^2) \dfrac{1}{r^2} + 2 \alpha E \dfrac{1}{r} + E^2 - m^2 c^4}} \, \dfrac{\dd r}{r^2}.\\
	t &= \dfrac{E}{c} \int \dfrac{r + \alpha E / (E^2 - m^2 c^4)}{\sqrt{(E^2 - m^2 c^4) r^2 + 2 \alpha E r + \alpha^2 - c^2 L^2}} \, \dd r - \dfrac{\alpha m^2 c^3}{E^2 - m^2 c^4} \int \dfrac{1}{\sqrt{(E^2 - m^2 c^4) r^2 + 2 \alpha E r + \alpha^2 - c^2 L^2}} \, \dd r.
\end{align*}

\begin{align*}
	r =
	\begin{cases}
		\dfrac{\alpha^2 - c^2 L^2}{-\alpha E + \sqrt{\alpha^2 E^2 - (E^2 - m^2 c^4) (\alpha^2 - c^2 L^2)} \cosh \left( \dfrac{\sqrt{\alpha^2 - c^2 L^2}}{c L} \phi \right)}, & \alpha > c L \\
		\dfrac{2 \alpha E}{E^2 \phi^2 - E^2 + m^2 c^4},                                                                                                                      & \alpha = c L \\
		\dfrac{-\alpha^2 + c^2 L^2}{\alpha E + \sqrt{\alpha^2 E^2 - (E^2 - m^2 c^4) (\alpha^2 - c^2 L^2)} \cos \left( \dfrac{\sqrt{-\alpha^2 + c^2 L^2}}{c L} \phi \right)}, & \alpha < c L
	\end{cases}
\end{align*}
\begin{note}
	\(\alpha=cL\) 时不存在有限的近日点；对正半径物理解，必须有
	\(\lvert\phi\rvert > \sqrt{E^2-m^2c^4}/E\)，等号处对应 \(r\to\infty\) 的渐近线，
	而 \(\phi=0\) 不属于该物理解域。因此这一临界分支的积分常数不能按近日点方向解释，也不能直接与另外两分支在同一角度零点下比较。
\end{note}

	\left| \phi \right|_{\min}

	\frac{

		}

	\frac{

\end{align*}

\begin{align*}
	r_{\max} =
	\frac{
		\alpha E +
		\sqrt{
			\alpha^2 E^2 - (E^2 - m^2 c^4)\,(\alpha^2 - c^2 L^2)
		}
	}{
		-m^2 c^4 + E^2
	},
\end{align*}

	r_{\min} =
	\frac{
		\alpha E -
		\sqrt{

		}
	}{
		-m^2 c^4 + E^2
	}.
\end{align*}

	\left| \phi \right|_{\max}

	\frac{cL}{\sqrt{-\alpha^2 + c^2 L^2}}

		\frac{

		}

\end{align*}

\begin{exbox}
	\(\quad\alpha < cL\) 且 \(E < mc^2\) 的情形，可用于处理开普勒椭圆轨道的狭义相对论修正.

	将\(r(\phi)\)写成
	\begin{align*}
		r = \dfrac{p}{1 + e \cos (\xi \phi)},
	\end{align*}
	其中
	\begin{align*}
		p = \dfrac{c^2 L^2 - \alpha^2}{\alpha E}, \quad
		e = \dfrac{\sqrt{\alpha^2 E^2 - (E^2 - m^2 c^4) (\alpha^2 - c^2 L^2)}}{\alpha E}, \quad
		\xi = \dfrac{\sqrt{c^2 L^2 - \alpha^2}}{cL}.
	\end{align*}

	进而质点的角向、径向广义动量为
	\begin{align*}
		p_{\phi} = L = \dfrac{m r^2 \dot{\phi}}{\sqrt{1 - v^2/c^2}}, \quad
		p_r = \dfrac{m}{\sqrt{1 - v^2/c^2}} \dfrac{\dd r}{\dd t} = -L \dfrac{\dd r^{-1}}{\dd \phi} = \dfrac{e\xi L}{p} \sin (\xi \phi).
	\end{align*}

	引入量子化条件
	\begin{align*}
		\oint p_{\phi} \, \dd \phi =\int_{0}^{2\pi} L \, \dd \phi = 2 \pi n_{\phi} \hbar, \quad
		\oint p_r \, \dd r =\int_{0}^{2\pi} \dfrac{e^2 \xi L \sin^2 (\xi \phi)}{(1 + e \cos (\xi \phi))^2} \, \dd(\xi \phi) = 2 \pi n_r \hbar.
	\end{align*}
	完成积分，解得
	\begin{align*}
		E = \left(1 + \left(\dfrac{\alpha}{\hbar c \left(n_r + \sqrt{n_{\phi}^2 - \left(\frac{\alpha}{\hbar c}\right)^2}\right)}\right)^2\,\right)^{-\frac{1}{2}} m c^2.
	\end{align*}

	对于氢原子
	\begin{align*}
		\alpha=\frac{e^2}{4\pi\varepsilon_0},\quad \frac{\alpha}{\hbar c}=\alpha_{fs}\approx\frac{1}{137}\,\text{\,(fs = fine structure)}
	\end{align*}
	记 \(n=n_r+n_{\phi}\)，则该相对论性半经典模型将氢原子结合能修正为
	\begin{align*}
		E_{\text{bound}} =
		E - mc^2 =
		-\frac{m c^2}{2 n^2}\,
		\alpha_{\mathrm{fs}}^2 \left(
		1 +
		\frac{
			\alpha_{\mathrm{fs}}^2
		}{n^2}\left(
		\frac{n}{n_{\phi}} - \frac{3}{4}
		\right)
		+ \mathcal{O}\bigl(\alpha_{\mathrm{fs}}^4\bigr)
		\right).
	\end{align*}
\end{exbox}

\begin{exbox}
	\(\quad\alpha < cL\) 且 \(E > mc^2\) 的情形，可用于处理开普勒双曲线轨道的狭义相对论修正

	记
	\begin{align*}
		\phi_{\infty} = |\phi|_{\text{min}} = \dfrac{cL}{\sqrt{-\alpha^2 + c^2 L^2}} \left(\pi - \arctan \dfrac{\sqrt{(E^2 - m^2 c^4) (-\alpha^2 + c^2 L^2)}}{\alpha E}\right).
	\end{align*}
	则物体由无穷远入射势场再出射到无穷远时，运动方向向力心将偏转角度
	\begin{align*}
		\chi = 2 \phi_{\infty} - \pi = \dfrac{2 cL}{\sqrt{-\alpha^2 + c^2 L^2}} \left(\pi - \arctan \dfrac{\sqrt{(E^2 - m^2 c^4) (-\alpha^2 + c^2 L^2)}}{\alpha E}\right) - \pi.
	\end{align*}
	设物体在无穷远处速率为 \(\beta_0 c\)，入射半径为 \(b\)，则：
	\begin{align*}
		E = \dfrac{mc^2}{\sqrt{1 - \beta_0^2}}, \quad
		L = \dfrac{mc \beta_0 b}{\sqrt{1 - \beta_0^2}}.
	\end{align*}
	记 \(\zeta = \dfrac{\alpha}{mc^2 b\, \beta_0^2}\)，可以化简得
	\begin{align*}
		\chi = -\pi + \dfrac{2 \pi}{\sqrt{1 - \beta_0^2 (1 - \beta_0^2) \zeta^2}} - \dfrac{2}{\sqrt{1 - \beta_0^2 (1 - \beta_0^2) \zeta^2}} \arctan \left(\zeta^{-1} \sqrt{\dfrac{1 - \beta_0^2 (1 - \beta_0^2) \zeta^2}{1 - \beta_0^2}}\right).
	\end{align*}

	关于 \(\beta_0\) 做小量展开
	\begin{align*}
		\chi = 2 \arctan\zeta + \left( \dfrac{\zeta (\zeta^2 - 1)}{1 + \zeta^2} + \zeta^2 \left( \dfrac{\pi}{2} + \arctan\zeta \right) \right) \beta_0^2 + \mathcal{O} (\beta_0^4).
	\end{align*}
	其中\(\beta_0^2\)项即为偏转角的相对论修正.
\end{exbox}

\section{在一次反比排斥势场中的运动}
假设空间中由固定源产生一次反比势能分布
\begin{align*}
	V(r) = \dfrac{\alpha}{r}, \quad \alpha > 0.
\end{align*}

设运动的作用量为 \(S\)，则哈密顿 - 雅可比方程给出：
\begin{align*}
	\left( \dfrac{\partial S}{\partial r} \right)^2 + \left( \dfrac{\partial S}{r \partial \phi} \right)^2 + m^2 c^2 = \dfrac{1}{c^2} \left( E - \dfrac{\alpha}{r} \right)^2.
\end{align*}
分离变量，设 \(S = -Et + L\phi + f(r)\)，得：
\begin{align*}
	\left( \dfrac{\dd f(r)}{\dd r} \right)^2 = \dfrac{1}{c^2} \left( E - \dfrac{\alpha}{r} \right)^2 - \dfrac{L^2}{r^2} - m^2 c^2.
\end{align*}
因此
\begin{align*}
	f(r) = \int \sqrt{\dfrac{1}{c^2} \left( E - \dfrac{\alpha}{r} \right)^2 - \dfrac{L^2}{r^2} - m^2 c^2} \, \dd r,\quad
	S = -Et + L\phi + \int \sqrt{\dfrac{1}{c^2} \left( E - \dfrac{\alpha}{r} \right)^2 - \dfrac{L^2}{r^2} - m^2 c^2} \, \dd r.
\end{align*}
代入正则方程，并适当选择角度和时间零点，使得常数为 0：
\begin{align*}
	\dfrac{\partial S}{\partial L} = \phi_0 = 0, \quad \dfrac{\partial S}{\partial E} = t_0 = 0.
\end{align*}
进而得到 \(\phi(r)\) 和 \(t(r)\) 的函数关系：
\begin{align*}
	\phi &= \int \dfrac{cL}{\sqrt{(\alpha^2 - c^2 L^2) \dfrac{1}{r^2} - 2 \alpha E \dfrac{1}{r} + E^2 - m^2 c^4}} \, \dfrac{\dd r}{r^2}.\\
	t &= \dfrac{E}{c} \int \dfrac{r - \alpha E / (E^2 - m^2 c^4)}{\sqrt{(E^2 - m^2 c^4) r^2 - 2 \alpha E r + \alpha^2 - c^2 L^2}} \, \dd r + \dfrac{\alpha m^2 c^3}{E^2 - m^2 c^4} \int \dfrac{1}{\sqrt{(E^2 - m^2 c^4) r^2 - 2 \alpha E r + \alpha^2 - c^2 L^2}} \, \dd r.
\end{align*}

运用公式，完成积分得：
\begin{equation*}
	\begin{split}
			t & = \dfrac{E}{c} \dfrac{1}{E^2 - m^2 c^4} \sqrt{(E^2 - m^2 c^4) r^2 - 2 \alpha E r + \alpha^2 - c^2 L^2} + \dfrac{\alpha m^2 c^4}{c (E^2 - m^2 c^4)^{3/2}} \arcosh \left( \dfrac{-\alpha E + r (E^2 - m^2 c^4)}{\sqrt{\alpha^2 E^2 - (E^2 - m^2 c^4) (\alpha^2 - c^2 L^2)}} \right) \\
			r & =
			\begin{cases}
				\dfrac{\alpha^2 - c^2 L^2}{\alpha E - \sqrt{\alpha^2 E^2 - (E^2 - m^2 c^4) (\alpha^2 - c^2 L^2)} \cosh \left( \dfrac{\sqrt{\alpha^2 - c^2 L^2}}{c L} \phi \right)},   & \alpha > c L \\
				\dfrac{2 \alpha E}{E^2 (1 - \phi^2) - m^2 c^4},                                                                                                                       & \alpha = c L \\
				\dfrac{-\alpha^2 + c^2 L^2}{-\alpha E + \sqrt{\alpha^2 E^2 - (E^2 - m^2 c^4) (\alpha^2 - c^2 L^2)} \cos \left( \dfrac{\sqrt{-\alpha^2 + c^2 L^2}}{c L} \phi \right)}, & \alpha < c L
			\end{cases}
		\end{split}
\end{equation*}
其中，由于总有 \(E > mc^2\)，\(t(r)\) 的关系是普适的，不必解析延拓.

质点到力心的距离总存在下限
\begin{align*}
	r_{\text{min}} = \dfrac{\alpha E + \sqrt{\alpha^2 E^2 - (E^2 - m^2 c^4) (\alpha^2 - c^2 L^2)}}{E^2 - m^2 c^4}.
\end{align*}

质点的入射角也总存在上限
\begin{align*}
	|\phi|_{\text{max}} =
	\begin{cases}
		\dfrac{c L}{\sqrt{\alpha^2 - c^2 L^2}} \text{arctanh} \dfrac{\sqrt{(E^2 - m^2 c^4) (\alpha^2 - c^2 L^2)}}{\alpha E}, & \alpha > c L \\[10pt]
		\sqrt{\dfrac{E^2 - m^2 c^4}{E^2}},                                                                                   & \alpha = c L \\[10pt]
		\dfrac{c L}{\sqrt{-\alpha^2 + c^2 L^2}} \arctan \dfrac{\sqrt{(E^2 - m^2 c^4) (-\alpha^2 + c^2 L^2)}}{\alpha E},      & \alpha < c L
	\end{cases}
\end{align*}

\begin{exbox}

	物体由无穷远入射势场再出射到无穷远时，运动方向向力心将偏转角度：
	\begin{align*}

\section{非相对论吸引势的微扰理论}
假设空间中由固定源产生略加修正的一次反比势能分布
\begin{align*}
	V(r) = -\dfrac{\alpha}{r} + \varepsilon(r, \theta),\quad |\varepsilon(r, \theta)| \ll \dfrac{\alpha}{r}.
\end{align*}

质量为 \(m\) 的粒子的拉普拉斯-龙格-楞次矢量为
\begin{align*}
	\vb{A} = \dfrac{\vb{v} \times \vb{L}}{\alpha} - \dfrac{\vb{r}}{r}.
\end{align*}
由运动方程：
\begin{align*}
	\dfrac{\dd \uv{r}}{\dd t} = \dfrac{\vb{L}}{m r^3} \times \vb{r}, \quad
	\vb{f} = m \dfrac{\dd \vb{v}}{\dd t}, \quad
	\dfrac{\dd \vb{L}}{\dd t} = \vb{r} \times \vb{f}.
\end{align*}
则
\begin{align*}
	\dfrac{\dd \vb{A}}{\dd t} = \left( \dfrac{\vb{f}}{\alpha} + \dfrac{\vb{r}}{r^3} \right) \times \dfrac{\vb{L}}{m} + \dfrac{\vb{v} \times (\vb{r} \times \vb{f})}{\alpha}.
\end{align*}

对于无微扰势的零级运动解
\begin{align*}
	\vb{f}^{(0)} = -\dfrac{\alpha}{r^3} \vb{r},\quad r=\frac{p}{1+e\cos\theta}
\end{align*}
\begin{align*}
	\quad L^2=m\alpha p,\quad \vb{v}=\frac{L}{mr}(\uv{\theta}+\frac{e\sin\theta}{1+e\cos\theta}\uv{r}),\quad \frac{\dd \theta}{\dd t}=\frac{L}{mr^2}
\end{align*}
代入知 \(\dd \vb{A}/\dd t=0\)，即\(\vb{A}^{(0)} = e\, (\uv{r} \cos\theta - \uv{\theta} \sin\theta)\)为常矢量，而引入微扰力将使\(\vb{A}\)随时间变化，进而给出轨道参量的修正.记 \(f_r = -\partial\varepsilon(r,\theta)/\partial r\)，\(f_{\theta} = -\partial\varepsilon(r,\theta)/r\partial\theta\)，代入 \(\vb{f} = -\dfrac{\alpha}{r^3} \vb{r} + f_r \uv{r} + f_{\theta} \uv{\theta}\,\) 得：
\begin{align*}
	\dfrac{\dd \vb{A}}{\dd t} = \left( \dfrac{L}{m \alpha} \right) \left[ 2 f_{\theta} \uv{r} - \left( f_r + \dfrac{e \sin\theta}{1 + e \cos\theta} f_{\theta} \right) \uv{\theta} \,\right]
\end{align*}
引入微扰势后，有垂直于 \(\vb{A}^{(0)}\) 的有效变化率：
\begin{align*}
	\dfrac{\dd A_{\text{eff}}}{\dd t} = \dfrac{\dd \vb{A}}{\dd t} \cdot (\uv{\theta} \cos\theta + \uv{r} \sin\theta) = \dfrac{L}{m \alpha} \left( -f_r \cos\theta + f_{\theta} \dfrac{\sin\theta\, (2 + e \cos\theta)}{1 + e \cos\theta} \right).
\end{align*}
在给定的运动范围\(D_{\theta}\)内，
\begin{align*}
	\Delta A_{\text{eff}} = \int_{D_{\theta}} \dfrac{\dd A_{\text{eff}}}{\dd t} \, \dd t = \int_{D_{\theta}} \dfrac{\dd A_{\text{eff}}}{\dd t} \sqrt{\dfrac{m p^3}{\alpha}} \, \dfrac{\dd \theta}{(1 + e \cos\theta)^2}.
\end{align*}
代入 \(L = \sqrt{m\alpha p}\)，因此轨道长轴在该运动范围内的进动角为：
\begin{align*}
	\Delta \phi = \dfrac{\Delta A_{\text{eff}}}{|\vb{A}^{(0)}|} = \dfrac{p^2}{\alpha e} \int_{D_{\theta}} \left( -f_r \dfrac{\cos\theta}{(1 + e \cos\theta)^2} + f_{\theta} \dfrac{(2 + e \cos\theta) \sin\theta}{(1 + e \cos\theta)^3} \right) \, \dd \theta
\end{align*}

对于椭圆轨道，\(e < 1\).

运动范围取为一周期，\(0 \leq \theta \leq 2\pi\)，得周期进动角为：
\begin{align*}
	\Delta \phi_{\text{pr}} = \dfrac{p^2}{\alpha e} \int_{0}^{2\pi} \left( -f_r \dfrac{\cos\theta}{(1 + e \cos\theta)^2} + f_{\theta} \dfrac{(2 + e \cos\theta) \sin\theta}{(1 + e \cos\theta)^3} \right) \, \dd \theta.
\end{align*}

对于双曲线轨道，\(e > 1\).

运动范围取入射到出射全过程，\(-\pi + \arccos \dfrac{1}{e} \leq \theta \leq \pi - \arccos \dfrac{1}{e}\)，得附加偏转角为：
\begin{align*}
	\Delta \chi = \dfrac{p^2}{\alpha e} \int_{-\pi + \arccos \frac{1}{e}}^{\pi - \arccos \frac{1}{e}} \left( -f_r \dfrac{\cos\theta}{(1 + e \cos\theta)^2} + f_{\theta} \dfrac{(2 + e \cos\theta) \sin\theta}{(1 + e \cos\theta)^3} \right) \, \dd \theta.
\end{align*}
附加上零级偏转角 \(\chi^{(0)} = \pi - 2 \arccos \dfrac{1}{e}\)，得运动的总偏转角为：
\begin{align*}
	\chi = \pi - 2 \arccos \frac{1}{e} + \dfrac{p^2}{\alpha e} \int_{-\pi + \arccos \frac{1}{e}}^{\pi - \arccos \frac{1}{e}} \left( -f_r \dfrac{\cos\theta}{(1 + e \cos\theta)^2} + f_{\theta} \dfrac{(2 + e \cos\theta) \sin\theta}{(1 + e \cos\theta)^3} \right) \, \dd \theta.
\end{align*}

\begin{exbox}
	广义相对论给出的修正势为
	\begin{align*}
		\varepsilon (r, \theta) = -\frac{\alpha  L^2}{m^2 c^2 r^3},
		\quad
		f_r = -\frac{\partial \varepsilon (r, \theta)}{\partial r} = -\frac{3 \alpha  L^2}{m^2 c^2 r^4} = -\frac{3 \alpha^2}{m p^3 c^2} (1 + e \cos  \theta)^4;
	\end{align*}
	代入公式得
	\begin{align*}
		\Delta\phi_{\text{pr}} = \frac{p^2}{\alpha e} \int_0^{2\pi} \left(
		-f_r \cdot \frac{\cos \theta}{(1 + e \cos \theta)^2}
		+
		f_\theta \cdot \frac{(2 + e \cos \theta) \sin \theta}{(1 + e \cos \theta)^3}
		\right) \dd \theta
		= \frac{6 \pi  \alpha}{m c^2 p};
	\end{align*}
	平均进动角速度为
	\begin{align*}
		\Omega_{\text{pr}} =
		\frac{\Delta\phi_{\text{pr}}}{T} =  \frac{\Delta\phi_{\text{pr}}}{2\pi}\sqrt{\frac{\alpha}{m p^3}}=
		\frac{3}{c^2} \sqrt{\frac{\alpha^3}{m^3 p^5}} (1 - e^2)^{3/2};
	\end{align*}

	若取 \(\alpha/mc^2 = G M_{\text{sun}}/mc^2 = 1.477 \times 10^3 \text{ m}\)，
	\(p = p_{\text{mercury}} = 5.546 \times 10^{10} \text{ m}\)，
	\(e = e_{\text{mercury}} = 0.2056\)，积分得
	\begin{align*}
		\Delta\phi_{\text{pr}} = 5.02 \times 10^{-7} \text{ rad}, \quad
		\Omega_{\text{pr}} = 43''/\text{century},
	\end{align*}
	即为著名的水星百年进动角.
\end{exbox}

\begin{exbox}
	对于长轴位于方位角 \(\alpha\) 处、半长轴为 \(a\)、半短轴为 \(b\) 的旋转长椭球，其修正势为

		- \arccos \left(\frac{1}{e} \right)
		+ \frac{2 (e^2 - 1)^{5/2} \cos (2 \alpha)}{3 e^4}
		\right).
	\end{align*}
\end{exbox}
