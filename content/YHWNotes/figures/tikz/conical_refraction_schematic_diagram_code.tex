\tdplotsetmaincoords{50}{20}
\begin{tdplot}[>=Stealth]
  \def\L{4}
  \def\psiDeg{25}
  \pgfmathsetmacro{\R}{\L*tan(\psiDeg)}
  \pgfmathsetmacro{\Cz}{\R}

  \foreach \phi in {0,45,...,315}{
    \draw[black!70, thick, ->]
      (0,0,0) -- ({\L}, {\R*sin(\phi)}, {\R*(1-cos(\phi))});
  }

  \draw[black!75, dashed]
    plot[domain=0:360, samples=121, variable=\phi]
    ({\L}, {\R*sin(\phi)}, {\Cz-\R*cos(\phi)});
\end{tdplot}
