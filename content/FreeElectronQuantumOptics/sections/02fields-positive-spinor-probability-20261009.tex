协变旋量用于散射外线，而实际束流由普通概率归一化的正能波包制备。两种归一化必须在进入跃迁核之前换算，因为概率投影、自旋求和与相空间使用的内积不同。

\begin{definition}[自由 Hamiltonian 与概率归一化旋量]
令 $\beta=\gamma^0$、$\alpha_i=\gamma^0\gamma^i$，则
\begin{align}
 \{\alpha_i,\alpha_j\}&=2\delta_{ij}I_4,\qquad
 \{\alpha_i,\beta\}=0,\qquad \beta^2=I_4,
 \label{eq:feqo-dirac-anticommutators}\\
 H_0(\mathbf p)&=c\boldsymbol\alpha\cdot\mathbf p+\beta m_{\mathrm e}c^2,
 \qquad H_0^2=E_{\mathbf p}^2I_4.
 \label{eq:feqo-dirac-free}
\end{align}
这里 $\alpha_i$ 与 $\beta$ 都厄米，$E_{\mathbf p}>0$。协变式和 Hamiltonian 式之间的词典是
\begin{equation}
 \gamma^0=\beta,\qquad \gamma^i=\beta\alpha_i.
 \label{eq:feqo-gamma-dictionary}
\end{equation}
将\eqnrefs{eq:explicit-u} 的旋量缩放为 $U_s(\mathbf p)=\sqrt{c/(2E_{\mathbf p})}\,u_s(p)$，便有
\begin{equation}
 U_s(\mathbf p)=N_{\mathbf p}
 \begin{pmatrix}\xi_s\\ C_{\mathbf p}\xi_s\end{pmatrix},
 \qquad
 N_{\mathbf p}=\sqrt{\frac{E_{\mathbf p}+m_{\mathrm e}c^2}{2E_{\mathbf p}}},
 \qquad
 C_{\mathbf p}=\frac{c\boldsymbol\sigma_{\rm P}\cdot\mathbf p}
 {E_{\mathbf p}+m_{\mathrm e}c^2}.
 \label{eq:feqo-positive-spinor}
\end{equation}
大写 $U_s$ 在本段表示单位概率范数旋量；小写 $u_s(p)$ 始终保留\eqnrefs{eq:explicit-u} 的协变归一化。
\end{definition}

\begin{proposition}[同一动量上的正交能量分解]
在固定 $\mathbf p$ 上，定义
\begin{equation}
 \Pi_\pm(\mathbf p)=\frac12\left(I_4\pm\frac{H_0(\mathbf p)}{E_{\mathbf p}}\right).
 \label{eq:feqo-hamiltonian-energy-projectors}
\end{equation}
则 $\Pi_\pm^2=\Pi_\pm=\Pi_\pm^\dagger$、$\Pi_+\Pi_-=0$、$\Pi_++\Pi_-=I_4$，且 $\Tr\Pi_\pm=2$、$H_0\Pi_\pm=\pm E_{\mathbf p}\Pi_\pm$。同一三动量上的负能本征旋量可取
\begin{equation}
 W_s(\mathbf p)=N_{\mathbf p}
 \begin{pmatrix}-C_{\mathbf p}\eta_s\\\eta_s\end{pmatrix},
 \qquad H_0(\mathbf p)W_s(\mathbf p)=-E_{\mathbf p}W_s(\mathbf p),
 \label{eq:feqo-negative-hamiltonian-spinor}
\end{equation}
其中 $\eta_s$ 是二分量单位正交基。$\{U_1,U_2,W_1,W_2\}$ 是普通内积下的完整正交基。
\end{proposition}

\begin{proof}
Dirac 本征方程的上下两块为
\begin{equation*}
 (E-m_{\mathrm e}c^2)\phi=c\boldsymbol\sigma_{\rm P}\cdot\mathbf p\,\chi,
 \qquad
 (E+m_{\mathrm e}c^2)\chi=c\boldsymbol\sigma_{\rm P}\cdot\mathbf p\,\phi.
\end{equation*}
正能根给 $\chi=C_{\mathbf p}\phi$；负能根给 $\phi=-C_{\mathbf p}\chi$。因为 $C_{\mathbf p}$ 厄米且
\begin{equation*}
 C_{\mathbf p}^2=
 \frac{E_{\mathbf p}-m_{\mathrm e}c^2}{E_{\mathbf p}+m_{\mathrm e}c^2}I_2,
 \qquad N_{\mathbf p}^2(I_2+C_{\mathbf p}^2)=I_2,
\end{equation*}
有 $U_s^\dagger U_{s'}=W_s^\dagger W_{s'}=\delta_{ss'}$。不同能量支的正交性也直接给出
\begin{equation*}
 U_s^\dagger W_{s'}=N_{\mathbf p}^2\xi_s^\dagger(-C_{\mathbf p}+C_{\mathbf p})\eta_{s'}=0.
\end{equation*}
令 $h=H_0/E_{\mathbf p}$，则 $h=h^\dagger$、$h^2=I_4$，从而 $(I_4\pm h)^2=2(I_4\pm h)$ 和 $(I_4+h)(I_4-h)=0$ 给出所有投影恒等式。$\Tr H_0=0$ 又给 $\Tr\Pi_\pm=2$。因此
\begin{equation*}
 \Pi_+=\sum_sU_sU_s^\dagger,
 \qquad \Pi_-=\sum_sW_sW_s^\dagger.
\end{equation*}
对于\eqnrefs{eq:explicit-v}，$H_0(-\mathbf p)v_s(p)=-E_{\mathbf p}v_s(p)$，故 $W_s(\mathbf p)=\sqrt{c/(2E_{\mathbf p})}\,v_s(p^0,-\mathbf p)$。负频反粒子旋量与同一三动量的负能 Hamiltonian 旋量不能使用同一个动量实参。
\end{proof}

\begin{proposition}[协变自旋和与概率投影的换算]
未来指向在壳动量 $p$ 对应
\begin{align}
 \sum_sU_s(\mathbf p)\bar U_s(\mathbf p)
 &=\Pi_+(\mathbf p)\gamma^0
 =\frac{c(\slashed p+m_{\mathrm e}c)}{2E_{\mathbf p}},
 \label{eq:feqo-probability-spin-sum}\\
 \sum_sW_s(\mathbf p)\bar W_s(\mathbf p)
 &=\Pi_-(\mathbf p)\gamma^0
 =\frac{c(\slashed{\tilde p}-m_{\mathrm e}c)}{2E_{\mathbf p}},
 \qquad \tilde p=(E_{\mathbf p}/c,-\mathbf p).
 \label{eq:feqo-negative-probability-spin-sum}
\end{align}
普通内积的正交投影为 $\Pi_\pm$；含 Dirac 共轭的自旋和并非普通正交投影。
\end{proposition}

\begin{proof}
右乘 $\gamma^0$ 时，$\alpha_i\gamma^0=-\gamma^i$，所以
\begin{equation*}
 \Pi_+\gamma^0=\frac{E_{\mathbf p}\gamma^0-c\gamma^ip_i+m_{\mathrm e}c^2I_4}{2E_{\mathbf p}},
 \qquad
 \Pi_-\gamma^0=\frac{E_{\mathbf p}\gamma^0+c\gamma^ip_i-m_{\mathrm e}c^2I_4}{2E_{\mathbf p}}.
\end{equation*}
两式分别是 $c(\slashed p+m_{\mathrm e}c)/(2E_{\mathbf p})$ 和 $c(\slashed{\tilde p}-m_{\mathrm e}c)/(2E_{\mathbf p})$。同时
\begin{equation*}
 \bar U_sU_{s'}=\frac{m_{\mathrm e}c^2}{E_{\mathbf p}}\delta_{ss'},
 \qquad
 \bar W_sW_{s'}=-\frac{m_{\mathrm e}c^2}{E_{\mathbf p}}\delta_{ss'}.
\end{equation*}
协变归一化只需乘回 $\sqrt{2E_{\mathbf p}/c}$，得到\eqnrefs{eq:u-complete,eq:v-complete}。
\end{proof}

\begin{exbox}[$200\,\mathrm{keV}$ 电子的上下分量与投影]
取 $\mathbf p=p\mathbf e_z$、$m_{\mathrm e}c^2=511\,\mathrm{keV}$，则总能量是 $E=711\,\mathrm{keV}$，
\begin{equation*}
 pc=\sqrt{711^2-511^2}\,\mathrm{keV}=494.4\,\mathrm{keV},
 \qquad r=\frac{pc}{E+m_{\mathrm e}c^2}=0.4046,
 \qquad N^2=\frac{1222}{1422}=0.85935.
\end{equation*}
自旋向上时 $U_\uparrow=N(1,0,r,0)^{\mathsf T}$，因此上下两块的概率分别为 $N^2=0.85935$ 和 $N^2r^2=0.14065$；下分量已占约 $14\%$。对坐标基向量 $e_1=(1,0,0,0)^{\mathsf T}$，
\begin{equation*}
 \Pi_+e_1=N^2(1,0,r,0)^{\mathsf T},
 \qquad \|\Pi_+e_1\|^2=N^2.
\end{equation*}
所以 $e_1$ 并非纯正能电子；归一化其正能部分才得到 $U_\uparrow$。同一旋量的流速为 $c^2p/E=0.6953c$，与精确色散的群速度相同。
\end{exbox}

\subsubsection{正能单电子波包与概率流}
\label{sec:feqo-current-and-packet}
\begin{definition}[正能单电子波包与概率流]
以普通三维 $\delta$ 归一化动量基写入射电子，
\begin{equation}
 \Psi(\mathbf r,t)=\sum_s\int\frac{\dd^3p}{(2\pi\hbar)^{3/2}}
 a_s(\mathbf p)U_s(\mathbf p)
 \ee^{\ii(\mathbf p\cdot\mathbf r-E_{\mathbf p}t)/\hbar},
 \qquad \sum_s\int\dd^3p\,|a_s(\mathbf p)|^2=1.
 \label{eq:feqo-positive-wavepacket}
\end{equation}
概率密度、概率流和电荷四流依次为
\begin{equation}
 \rho=\Psi^\dagger\Psi,\qquad
 \mathbf j_{\rm prob}=c\Psi^\dagger\boldsymbol\alpha\Psi,\qquad
 j_{\rm ch}^\mu=q_{\mathrm e}c\bar\Psi\gamma^\mu\Psi
 =(q_{\mathrm e}c\rho,q_{\mathrm e}\mathbf j_{\rm prob}).
 \label{eq:feqo-dirac-current}
\end{equation}
$[\rho]=\mathrm{m^{-3}}$、$[\mathbf j_{\rm prob}]=\mathrm{m^{-2}s^{-1}}$；进入 Maxwell 方程的是电荷流。
\end{definition}

\begin{proof}[证明（归一化、连续性与群速度）]
对位置内积先积分平面波，得到 $(2\pi\hbar)^3\delta^{(3)}(\mathbf p-\mathbf q)$。在分布支撑上 $U_s^\dagger(\mathbf p)U_{s'}(\mathbf p)=\delta_{ss'}$，因此
\begin{equation*}
 \int\dd^3r\,\Psi^\dagger\Psi
 =\sum_s\int\dd^3p\,|a_s(\mathbf p)|^2=1.
\end{equation*}
将 Dirac Hamiltonian 方程与其厄米共轭相加，质量项和实标势相消，微分动能给
\begin{equation*}
 \partial_t(\Psi^\dagger\Psi)
 =-c\big[(\partial_i\Psi^\dagger)\alpha_i\Psi
 +\Psi^\dagger\alpha_i\partial_i\Psi\big]
 =-\nabla\cdot\mathbf j_{\rm prob}.
\end{equation*}
波包衰减使无穷远通量消失，所以总概率守恒。对固定动量的正能旋量，Pauli 反对易关系直接给
\begin{align*}
 U_s^\dagger\alpha_iU_s
 &=N_{\mathbf p}^2\xi_s^\dagger(\sigma_iC_{\mathbf p}+C_{\mathbf p}\sigma_i)\xi_s\\
 &=N_{\mathbf p}^2\frac{2cp_i}{E_{\mathbf p}+m_{\mathrm e}c^2}
 =\frac{cp_i}{E_{\mathbf p}}.
\end{align*}
故 $\mathbf j_{\rm prob}/\rho=c^2\mathbf p/E_{\mathbf p}=\nabla_{\mathbf p}E$。波包中不同动量还存在干涉流，不能把该单平面波比值逐点代替完整电流。
\end{proof}
