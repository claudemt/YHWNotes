\section{计数似然与可识别性：从测量结果能恢复什么}
\label{chap:feqo-inversion}

\subsection{反问题的提出}

\cref{chap:feqo-measurement} 建立了从电子态到测量概率的前向映射：给定密度矩阵 $\rho$ 和测量设置（POVM $\{M_y\}$），概率为 $p(y|\rho)=\operatorname{Tr}(\rho M_y)$。\cref{chap:feqo-attosecond,chap:feqo-decoherence} 进一步给出了物理参数（耦合强度 $|\beta|$、退相干率 $D$ 等）如何决定 $\rho$。

现在问反问题：给定 $N$ 次测量的计数结果 $\{n_i\}$，能推断出什么物理参数？更精确地：

\begin{enumerate}
\item \textbf{似然函数}：参数 $\vartheta$ 产生观测数据 $\{n_i\}$ 的概率 $\mathcal{L}(\vartheta|\{n_i\})$ 是什么？
\item \textbf{精度极限}：任何无偏估计量的方差下界是多少？（Cram\'er--Rao 界）
\item \textbf{可识别性}：不同参数值是否产生不同的概率分布？若 $\vartheta_1\neq\vartheta_2$ 但 $p(y|\vartheta_1)=p(y|\vartheta_2)$，则数据无法区分两者。
\end{enumerate}

\subsection{计数问题的似然函数}

\subsubsection{Poisson 计数模型}

设实验有 $S$ 个测量设置（如不同参考相位 $\theta_s$），每个设置下有 $J$ 个探测通道（如能量边带 $\ell=0,\ldots,J-1$）。设置 $s$ 下通道 $j$ 的预期计数为
\begin{equation}
\mu_{js}(\vartheta)=\nu_s\, p_j(\vartheta,s),
\label{eq:feqo-expected-counts}
\end{equation}
其中 $\nu_s$ 是设置 $s$ 的入射电子数，$p_j(\vartheta,s)$ 是单电子在通道 $j$ 被检出的概率（由 POVM 框架给出）。

若各通道的计数独立（散粒噪声极限），$N_{js}$ 服从 Poisson 分布：
\begin{equation}
P(N_{js}|\mu_{js})=\frac{\mu_{js}^{N_{js}}e^{-\mu_{js}}}{N_{js}!}.
\label{eq:feqo-poisson-prob}
\end{equation}
对数似然为
\begin{equation}
\log\mathcal{L}(\vartheta)=\sum_{j=1}^{J}\sum_{s=1}^{S}
\bigl[N_{js}\log\mu_{js}(\vartheta)-\mu_{js}(\vartheta)-\log(N_{js}!)\bigr].
\label{eq:feqo-count-likelihood}
\end{equation}
最后一项 $\log(N_{js}!)$ 不依赖 $\vartheta$，在优化和求导时可丢弃。

\subsubsection{与 POVM 框架的联系}

\cref{eq:feqo-energy-povm} 给出 $p(y|\rho)=\sum_\ell R(y|E_\ell)\rho_{\ell\ell}$。若设置 $s$ 包含一个已知酉变换 $U_s$（如参考场相位旋转），则
\begin{equation}
p_j(\vartheta,s)=\operatorname{Tr}(U_s\rho(\vartheta)U_s^\dagger M_j)
=\sum_{\ell,\ell'}(U_s)_{j\ell}\rho_{\ell\ell'}(\vartheta)(U_s^\dagger)_{\ell'j}R(E_j|E_{\ell'}).
\label{eq:feqo-phase-readout-likelihood}
\end{equation}
因此似然函数通过 $\rho(\vartheta)$ 和 POVM $\{M_j\}$ 与量子测量理论完全连接。

\subsubsection{多项分布模型}

若总计数 $N_s=\sum_j N_{js}$ 固定（而非 Poisson 涨落），应改用多项分布：
\begin{equation}
P(\{N_{js}\}|\{p_{js}\})=\frac{N_s!}{\prod_j N_{js}!}\prod_j p_{js}^{N_{js}}.
\label{eq:feqo-multinomial}
\end{equation}
对数似然为
\begin{equation}
\log\mathcal{L}(\vartheta)=\sum_s\left[\sum_j N_{js}\log p_j(\vartheta,s)+\log\frac{N_s!}{\prod_j N_{js}!}\right].
\label{eq:feqo-multinomial-likelihood}
\end{equation}
两种模型在 $N_s$ 大时给出相近的 Fisher 信息（\cref{eq:feqo-poisson-fisher}），但多项分布更正确地描述了固定总计数的实验。

\subsection{Fisher 信息与 Cram\'er--Rao 界}

\subsubsection{Fisher 信息的定义}

定义得分（score）为对数似然对参数的导数：
\begin{equation}
S_a(\vartheta):=\frac{\partial}{\partial\vartheta^a}\log\mathcal{L}(\vartheta).
\label{eq:feqo-score}
\end{equation}
Fisher 信息矩阵为得分的外积期望：
\begin{equation}
I_{ab}(\vartheta):=\mathbb{E}_\vartheta[S_a S_b].
\label{eq:feqo-fisher-def}
\end{equation}
等价地（在正则条件下），$I_{ab}=-\mathbb{E}[\partial^2\log\mathcal{L}/\partial\vartheta^a\partial\vartheta^b]$。

\subsubsection{Poisson 计数的 Fisher 信息}

由\cref{eq:feqo-count-likelihood}，$\partial_a\log\mathcal{L}=\sum_{j,s}(N_{js}/\mu_{js}-1)\partial_a\mu_{js}$。利用 $\mathbb{E}N_{js}=\mu_{js}$，$\operatorname{Var}N_{js}=\mu_{js}$，独立性：
\begin{equation}
I_{ab}(\vartheta)=\sum_{j,s}\frac{\partial_a\mu_{js}(\vartheta)\,\partial_b\mu_{js}(\vartheta)}{\mu_{js}(\vartheta)}.
\label{eq:feqo-poisson-fisher}
\end{equation}
这一式也可以写成矩阵形式：令 $J$ 为 Jacobian 矩阵，$J_{(j,s),a}=\partial_a\mu_{js}$，$W$ 为对角权重矩阵，$W_{(j,s),(j,s)}=1/\mu_{js}$，则 $I=J^\top W J$。

\subsubsection{Cram\'er--Rao 界}

\begin{proposition}[计数实验的 Cram\'er--Rao 界]
\label{prop:cramer-rao}
设参数 $\vartheta$ 在开区间中，离散结果 $x$ 的概率 $P_\vartheta(x)>0$ 可微，且可把有限或绝对收敛的求和与求导交换。记得分 $S_\vartheta(x)=\partial_\vartheta\log P_\vartheta(x)$ 和信息 $I(\vartheta)=\mathbb{E}_\vartheta S_\vartheta^2>0$。若估计量 $T(x)$ 在此区间无偏，即 $\mathbb{E}_\vartheta T=\vartheta$，且方差有限，则
\begin{equation}
\operatorname{Var}_\vartheta T\geq I(\vartheta)^{-1}.
\label{eq:feqo-cr-bound}
\end{equation}
\end{proposition}
\begin{proof}
由概率归一化求导，$\mathbb{E}S_\vartheta=\sum_x\partial_\vartheta P_\vartheta(x)=0$。由无偏条件求导并用 $\partial_\vartheta P=P S_\vartheta$：
\begin{equation}
1=\frac{\dd}{\dd\vartheta}\mathbb{E}_\vartheta[T]=\sum_x T(x)\partial_\vartheta P_\vartheta(x)=\sum_x T(x)P_\vartheta(x)S_\vartheta(x)=\mathbb{E}[(T-\vartheta)S_\vartheta].
\end{equation}
（最后一步使用了 $\mathbb{E}[S_\vartheta]=0$，故 $\mathbb{E}[\vartheta S_\vartheta]=0$。）对随机变量 $T-\vartheta$ 与 $S_\vartheta$ 用 Cauchy--Schwarz 不等式：
\begin{equation}
1=(\mathbb{E}[(T-\vartheta)S_\vartheta])^2\leq\mathbb{E}[(T-\vartheta)^2]\cdot\mathbb{E}[S_\vartheta^2]=\operatorname{Var}T\cdot I(\vartheta).
\end{equation}
除以 $I(\vartheta)$ 即证。若 $P_\vartheta(x)=0$ 或参数处在边界，需要另行检查正则条件。
\end{proof}

\subsubsection{多参数 Cram\'er--Rao 界}

对多参数 $\vartheta=(\vartheta^1,\ldots,\vartheta^p)$，信息矩阵 $I_{ab}=\mathbb{E}[S_aS_b]$。在同样正则条件下，对无偏估计列向量 $T$ 求导可得 $\mathbb{E}[(T-\vartheta)S^\top]=\mathbf{1}$（单位矩阵）。若 $I$ 可逆，随机向量 $Z=T-\vartheta-I^{-1}S$ 的协方差矩阵半正定：
\begin{equation}
\operatorname{Cov}(T)-I^{-1}=\operatorname{Cov}(Z)\succeq 0.
\label{eq:feqo-cr-matrix}
\end{equation}
这里 $A\succeq 0$ 的定义是对每个实向量 $u$ 都有 $u^\top Au\geq 0$。矩阵界意味着对任意线性组合 $c^\top\vartheta$，$\operatorname{Var}(c^\top T)\geq c^\top I^{-1}c$。

\begin{note}[信息矩阵不可逆的含义]
若 $I$ 不可逆（秩不足），存在方向 $u\neq 0$ 使 $Iu=0$，即 $u^\top Iu=0$。这意味着数据对参数组合 $u^\top\vartheta$ 的一阶变化完全不敏感——该方向不可识别。此时不能对不可识别方向声称有限误差界。

\end{note}

\subsection{可识别性分析}

\subsubsection{全局与局部可识别性}

\begin{definition}[可识别性]
参数 $\vartheta$ 到结果概率 $p_j(\vartheta,s)$ 的映射，若不同参数始终产生不同的所有设置概率，称\emph{全局可识别}：$\vartheta_1\neq\vartheta_2\Rightarrow\exists(j,s):p_j(\vartheta_1,s)\neq p_j(\vartheta_2,s)$。其 Jacobian $J_{(j,s),a}=\partial p_j(\vartheta,s)/\partial\vartheta^a$ 满列秩是\emph{局部一阶可识别}的充分条件。
\end{definition}

秩不足时存在参数微扰 $u\neq 0$ 使 $Ju=0$，数据对这一方向的一阶变化为零。

\subsubsection{两能级例子：相位为什么需要参考读出}

设两边带电子态为
\begin{equation}
\rho(r,\phi)=\frac{1}{2}
\begin{pmatrix}1&r e^{-\ii\phi}\\r e^{\ii\phi}&1\end{pmatrix},
\qquad 0\leq r\leq 1.
\label{eq:feqo-two-level-state}
\end{equation}
两个本征值是 $(1\pm r)/2$，故这确是正、迹为一的密度矩阵。


\begin{exbox}[两能级相位的可识别性]
能量投影 $P_0=|0\rangle\langle 0|$，$P_1=|1\rangle\langle 1|$ 给出 $p_0=p_1=1/2$，对任意 $r,\phi$ 都相同。无论重复多少次，单一能谱都无法恢复相位 $\phi$。


增加投影到 $|+_\theta\rangle=(|0\rangle+e^{\ii\theta}|1\rangle)/\sqrt{2}$ 的参考设置，则
\begin{equation}
p_+(\theta)=\langle+_\theta|\rho|+_\theta\rangle
=\frac{1+r\cos(\phi-\theta)}{2}.
\label{eq:feqo-reference-phase-probability}
\end{equation}
取 $\theta=0,\pi/2$，两次测量分别给出 $r\cos\phi$ 与 $r\sin\phi$，从而在 $r>0$ 时恢复 $r,\phi$。当 $r=0$ 时相位根本没有物理定义，任意测量都不能"恢复"它。


\par
\begin{workedexample}
对\cref{eq:feqo-reference-phase-probability} 求 $\theta=0,\pi/2$ 时关于 $(r,\phi)$ 的 Jacobian 行列式，说明它在 $r=0$ 为何为零，并解释其物理含义。
\textbf{解.} 两个概率对 $(r,\phi)$ 的导数矩阵为 $J=\tfrac12\begin{psmallmatrix}\cos\phi&-r\sin\phi\\\sin\phi&r\cos\phi\end{psmallmatrix}$，故 $\det J=r/4$。$r=0$ 时相干项消失，概率不依赖相位；这时相位不是可识别参数，Jacobian 退秩正是这一物理事实的表达。
\end{workedexample}
\par

两次"$+$"概率记为 $p_0=(1+r\cos\phi)/2$，$p_{\pi/2}=(1+r\sin\phi)/2$。它们对 $(r,\phi)$ 的导数矩阵为
\begin{equation}
J=\frac{1}{2}
\begin{pmatrix}
\cos\phi&-r\sin\phi\\
\sin\phi&r\cos\phi
\end{pmatrix},
\qquad \det J=\frac{r}{4}.
\label{eq:feqo-jacobian-two-level}
\end{equation}
$r>0$ 时 Jacobian 满秩，局部可识别。$r=0$ 时 $\det J=0$，不可识别——与物理分析一致。

\end{exbox}

\subsubsection{可识别性的物理意义}

可识别性不是实验精度的问题，而是\emph{信息结构}的问题：前向映射是否把不同参数映射到不同的可观测分布。增加测量设置（如参考相位）可以补上前向映射缺少的独立方向，使不可识别的参数变得可识别。

\subsection{算例：从光子计数估计耦合强度 $g$}

\subsubsection{问题设定}

设 PINEM 实验中，激光波长和电子能量已知，要估计的参数是耦合强度 $|\beta|$（记作 $g$）。测量设置固定为单一能谱（无参考场），探测通道为边带 $\ell=0,\pm 1,\pm 2,\ldots$。

单电子概率为 $p_\ell(g)=J_\ell^2(2g)$。设入射电子数为 $N$，预期计数 $\mu_\ell(g)=N J_\ell^2(2g)$。

\subsubsection{Fisher 信息的计算}

由\cref{eq:feqo-poisson-fisher}：
\begin{equation}
I(g)=\sum_\ell\frac{(\partial_g\mu_\ell)^2}{\mu_\ell}
=N\sum_\ell\frac{(\partial_g J_\ell^2(2g))^2}{J_\ell^2(2g)}.
\label{eq:feqo-fisher-g}
\end{equation}
利用 $\partial_g J_\ell^2(2g)=2J_\ell(2g)\cdot 2J_\ell'(2g)=4J_\ell(2g)J_\ell'(2g)$：
\begin{equation}
I(g)=N\sum_\ell\frac{16J_\ell^2(2g)J_\ell'^2(2g)}{J_\ell^2(2g)}
=16N\sum_\ell J_\ell'^2(2g).
\label{eq:feqo-fisher-g-simplified}
\end{equation}
利用 Bessel 递推关系 $2J_\ell'(x)=J_{\ell-1}(x)-J_{\ell+1}(x)$：
\begin{equation}
J_\ell'(2g)=\frac{1}{2}[J_{\ell-1}(2g)-J_{\ell+1}(2g)].
\label{eq:feqo-bessel-derivative}
\end{equation}
代入：
\begin{equation}
I(g)=16N\sum_\ell\frac{1}{4}[J_{\ell-1}(2g)-J_{\ell+1}(2g)]^2
=4N\sum_\ell[J_{\ell-1}^2-2J_{\ell-1}J_{\ell+1}+J_{\ell+1}^2].
\label{eq:feqo-fisher-g-expanded}
\end{equation}
利用 $\sum_\ell J_\ell^2=1$ 和 $\sum_\ell J_{\ell-1}J_{\ell+1}=0$（平移正交性，因为 $J_{\ell-1}$ 和 $J_{\ell+1}$ 相差平移 2，而 $\sum J_\ell J_{\ell+k}=\delta_{k0}$）：
\begin{equation}
I(g)=4N(1+1-0)=8N.
\label{eq:feqo-fisher-g-final}
\end{equation}
等一下，让我重新检查。$\sum_\ell J_{\ell-1}^2=\sum_k J_k^2=1$，$\sum_\ell J_{\ell+1}^2=\sum_k J_k^2=1$，$\sum_\ell J_{\ell-1}J_{\ell+1}=\sum_k J_k J_{k+2}=0$（Neumann 加法定理的推论）。因此
\begin{equation}
I(g)=4N(1-0+1)=8N.
\label{eq:feqo-fisher-g-result}
\end{equation}

\begin{note}[Fisher 信息的含义]
Fisher 信息 $I(g)=8N$ 与 $g$ 无关！这意味着估计 $g$ 的精度只依赖于总电子数 $N$，不依赖于耦合强度本身。Cram\'er--Rao 界给出
\begin{equation}
\operatorname{Var}\hat{g}\geq\frac{1}{8N}.
\label{eq:feqo-cr-g}
\end{equation}
标准差 $\sigma_g\geq 1/\sqrt{8N}$。对 $N=10^4$，$\sigma_g\geq 0.0035$。

\end{note}

\subsubsection{弱耦合极限}

当 $g\ll 1$ 时，只有 $\ell=0,\pm 1$ 有显著概率：$p_0\approx 1-g^2$，$p_{\pm 1}\approx g^2$。Fisher 信息近似为
\begin{equation}
I(g)\approx N\left[\frac{(-2g)^2}{1-g^2}+2\cdot\frac{(2g)^2}{g^2}\right]
\approx N(4g^2+8)\approx 8N.
\label{eq:feqo-fisher-weak-g}
\end{equation}
与精确结果一致。

\subsubsection{强耦合时的截断效应}

实际实验中，只能探测有限范围的边带 $|\ell|\leq L$。若 $2g>L$，部分概率落在探测窗口外，Fisher 信息减少。截断后的信息为
\begin{equation}
I_{\rm trunc}(g)=4N\sum_{|\ell|\leq L}[J_{\ell-1}^2(2g)-2J_{\ell-1}(2g)J_{\ell+1}(2g)+J_{\ell+1}^2(2g)].
\label{eq:feqo-fisher-truncated}
\end{equation}
当 $2g\gg L$ 时，$I_{\rm trunc}<8N$，估计精度下降。

\subsection{退相干对 Fisher 信息的影响}

\subsubsection{退相干如何改变概率分布}

\cref{chap:feqo-decoherence} 表明，退相干使密度矩阵的非对角元衰减：$\rho_{\ell\ell'}\to\rho_{\ell\ell'}D_{\ell\ell'}$。对能谱测量（POVM 在能量基对角），退相干不改变概率 $p_\ell=\rho_{\ell\ell}$——因为能谱只读对角元。

因此，如果只用单一能谱估计 $g$，退相干不影响 Fisher 信息。

\subsubsection{退相干对相位估计的影响}

若要估计的参数包含相位（如 $\arg\beta$ 或参考场相位），则需要非对角测量（如参考读出）。退相干压低非对角元，从而降低相位灵敏度。

以\cref{eq:feqo-reference-phase-probability} 为例，退相干使 $r\to r\cdot D$（$D<1$），概率变为 $p_+(\theta)=(1+rD\cos(\phi-\theta))/2$。Fisher 信息对 $\phi$ 的部分为
\begin{equation}
I_{\phi\phi}=\frac{N(rD)^2}{1-(rD)^2/4}\approx Nr^2D^2\quad(rD\ll 1).
\label{eq:feqo-fisher-phase-decoherence}
\end{equation}
与无退相干（$D=1$）相比，信息按 $D^2$ 衰减。Cram\'er--Rao 界 $\operatorname{Var}\hat\phi\gtrsim 1/(Nr^2D^2)$ 随退相干增强而恶化。


\par
\begin{workedexample}
对\cref{eq:feqo-fisher-phase-decoherence}，设退相干因子 $D=e^{-\Gamma t}$（$\Gamma$ 为退相干率），推导相位估计的 Cram\'er--Rao 界随传播时间 $t$ 的变化，并求使 $\operatorname{Var}\hat\phi$ 最小的最优测量时间。
\textbf{解.} 在 $rD\ll1$ 的近似下，$I_{\phi\phi}\simeq Nr^2e^{-2\Gamma t}$，所以 $\operatorname{Var}\hat\phi\ge e^{2\Gamma t}/(Nr^2)$。若样品信号与计数预算不随 $t$ 增长，界随时间单调变差，最优是在退相干前尽早测量（$t=0$）；若传播时间同时带来可用信号增长，则须把该增长写入 Fisher 信息后再优化。
\end{workedexample}
\par
\subsubsection{退相干与可识别性的关系}

极端情况下，$D=0$（完全退相干），$p_+(\theta)=1/2$ 对所有 $\theta$，信息 $I_{\phi\phi}=0$，相位完全不可识别。这与\cref{chap:feqo-decoherence}的结论一致：完全退相干使时间密度变平坦，所有相位信息丧失。


\par
\begin{workedexample}
设探测窗口只覆盖 $|\ell|\leq 2$，$g=3$。计算截断后的 Fisher 信息 $I_{\rm trunc}(g)$，并与完整信息 $8N$ 比较。说明窗口截断如何降低参数估计精度。
\textbf{解.} 将 $x=2g=6$ 代入\cref{eq:feqo-fisher-truncated}，有限和为 $I_{\rm trunc}\approx3.6896N$，约为完整值 $8N$ 的 $0.4612$。因此方差下界由 $1/(8N)$ 增至至少 $1/(3.6896N)$；窗口漏掉的边带含有关于 $g$ 的变化信息。
\end{workedexample}
\par
\subsection{正则化与偏差--方差权衡}

\subsubsection{小奇异值的问题}

即使 Jacobian 满秩，极小奇异值也会使噪声被放大。对线性化模型 $y=Ax+\epsilon$，令 $W$ 为噪声权重，$Lx$ 为希望平滑或压小的量。Tikhonov 估计最小化 $\|Ax-y\|_W^2+\lambda\|Lx\|^2$，其一阶条件为
\begin{equation}
(A^\dagger WA+\lambda L^\dagger L)\widehat{x}_\lambda=A^\dagger Wy.
\label{eq:feqo-tikhonov}
\end{equation}
在无噪声 $y=Ax_*$ 下，只要左矩阵可逆，偏差为
\begin{equation}
\widehat{x}_\lambda-x_*=-\lambda(A^\dagger WA+\lambda L^\dagger L)^{-1}L^\dagger Lx_*.
\label{eq:feqo-tikhonov-bias}
\end{equation}
正则化稳定小奇异值，也引入可计算偏差；$\lambda$ 应由独立验证数据、预测风险或明确的先验选择，不能称作物理退相干率。

\subsubsection{与退相干的区分}

正则化是数据分析的选择，不是物理过程。退相干是电子与环境相互作用导致的物理信息丢失。两者都使反问题更难，但来源不同：
\begin{itemize}
\item 退相干减少 Fisher 信息（数据本身包含的信息变少）。
\item 正则化引入偏差（分析者主动牺牲无偏性以换取稳定性）。
\end{itemize}
在报告结果时，应分别说明退相干的物理限制和正则化的分析选择。

\subsection{本章小结}

计数似然通过 POVM 与量子测量框架连接。Poisson 计数的 Fisher 信息见\cref{eq:feqo-poisson-fisher}，它给出 Cram\'er--Rao 精度下界。可识别性要求 Jacobian 满列秩，也就是不同参数必须产生不同的可观测分布。对 PINEM 耦合强度 $g$，本章算得 $I(g)=8N$，与 $g$ 无关。

退相干不改变能谱的对角元，却会压低非对角测量对相位的灵敏度，使 Fisher 信息按 $D^2$ 衰减。
