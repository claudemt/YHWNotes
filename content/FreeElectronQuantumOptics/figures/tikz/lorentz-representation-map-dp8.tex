\begin{tikzpicture}[x=1cm,y=1cm,>=Stealth,
 every node/.style={font=\small},
 box/.style={draw,rounded corners=2pt,inner xsep=7pt,inner ysep=4pt,align=center,line width=.75pt},
 edge/.style={->,line width=.75pt},
 line label/.style={fill=white,fill opacity=.96,text opacity=1,inner sep=1.2pt}]
% Group-level relation: SL(2,C) maps two-to-one onto the proper orthochronous Lorentz group.
\node[box] (so) at (-3.7,1.35) {$SO^+(1,3)$\\proper orthochronous\\Lorentz group};
\node[box] (sl) at (1.8,1.35) {$SL(2,\mathbb C)$};
\draw[edge] (sl.west)--node[line label,above]{2:1 covering map}(so.east);

% The two inequivalent fundamental chiral representations are representations of SL(2,C).
\node[box] (L) at (-1.25,-1.0) {$(\frac12,0)$\\left-handed Weyl spinor $L$};
\node[box] (R) at (4.85,-1.0) {$(0,\frac12)$\\right-handed Weyl spinor $R$};
\draw[edge] (sl.south west) --node[line label,left]{representation}(L.north);
\draw[edge] (sl.south east) --node[line label,right]{representation}(R.north);

% A separate operation node prevents arrows from falsely suggesting particle conversion.
\node[box,minimum width=4.0cm] (build) at (1.8,-2.75)
 {build new Lorentz representations\\from $L$ and $R$};
\draw[edge] (L.south east) -- (build.north west);
\draw[edge] (R.south west) -- (build.north east);

\node[box,minimum width=3.7cm] (D) at (-3.9,-5.15)
  {$L\oplus R$\\Dirac spinor};
\node[box,minimum width=3.7cm] (V) at (1.8,-5.15)
  {$L\otimes R$\\four-vector $A^\mu$};
\node[box,minimum width=4.3cm] (F) at (7.2,-5.15)
  {$\mathrm{Sym}^2L\oplus\mathrm{Sym}^2R$\\antisymmetric tensor $F^{\mu\nu}$};
\draw[edge] (build.south west) to[out=225,in=70]
 node[line label,midway,left]{$\oplus$} (D.north east);
\draw[edge] (build.south) --node[line label,right]{$\otimes$}(V.north);
\draw[edge] (build.south east) to[out=-45,in=110]
 node[line label,midway,right]{$\mathrm{Sym}^2$} (F.north west);
\end{tikzpicture}
