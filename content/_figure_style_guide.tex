% =====================================================================
%  Block-logic / grid / map / flow TikZ conventions (input-able fragment)
% ---------------------------------------------------------------------
%  This file documents and tunes the shared block-logic grammar used by
%  relationship maps, layer diagrams and flow schematics across all books.
%
%  The base styles live in the root preamble.tex `\tikzset` block and are
%  always loaded:
%
%    lecturebox     rounded-corner content box (reuse this look)
%    lecturearrow   -{Stealth} arrow, 0.75 pt
%    line-label     white-backed label so lines never show through glyphs
%    blockbox       = lecturebox + minimum height (size a layer with
%                    `minimum width=<w>mm`, identical for every box in a row)
%    layerarrow     = lecturearrow, but ONLY horizontal/vertical routes
%    blocklabel     = line-label (alias)
%    blockmatrix    = `matrix of nodes` with nodes=blockbox, 10 mm gaps
%
%  Checklist (also in _figure_style_guide.md):
%   * Boxes: one minimum width per layer; use blockbox; no rotation;
%            explicit grid (matrix of nodes or fixed x/y).
%   * Arrows: horizontal flow = horizontal arrows; vertical flow = vertical
%             arrows.  Fan-out is L-shaped: `-|` (horizontal then vertical)
%             or `|-` (vertical then horizontal).  No out=/in= curves.
%   * Labels: always blocklabel/line-label, placed in whitespace beside the
%             arrow (above/below for horizontal, left/right for vertical),
%             never on the arrow root or on a box edge.
%   * Spacing: 8-10 mm column gap (or 3-4 cm explicit x-separation);
%             >= 1 cm padding around the bounding box.
%   * Overall: flat row baselines, vertical column edges, orthogonal lines.
%
%  Tuning below only sets figure-scoped defaults on top of preamble.
% =====================================================================
\tikzset{
  % Uniform gap and padding for block-logic grids built with blockmatrix.
  blockmatrix/.append style={column sep=9mm, row sep=9mm},
  % Slightly larger white backing for math arrow labels in wide maps.
  blocklabel/.append style={inner sep=1.4pt}
}
