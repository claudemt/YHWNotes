Fharmonic[\[Theta]_, \[Beta]_] := 
  Sin[\[Theta]]^2 (
   4 + \[Beta]^2 Cos[\[Theta]]^2)/(1 - \[Beta]^2 Cos[\[Theta]]^2)^(
   7/2);
Fcircle[\[Theta]_, \[Beta]_] := (
  4 (1 + Sin[\[Theta]]^2) - \[Beta]^2 Cos[\[Theta]]^4 (1 + 
      3 \[Beta]^2))/(1 - \[Beta]^2 Cos[\[Theta]]^2)^(7/2);
